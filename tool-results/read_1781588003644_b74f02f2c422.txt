   152→  const getAddressFromCoordinates = async (lat: number, lon: number): Promise<string> => {

   166→  const formatAddress = (data: any, useFullAddress: boolean): string => {
   167→    if (!data || !data.address) return ''
   168→    const addr = data.address
   169→    const parts: string[] = []
   170→
   171→    if (useFullAddress) {
   172→      if (addr.road) {
   173→        if (addr.house_number) {
   174→          parts.push(`${addr.road} ${addr.house_number}`)
   175→        } else {
   176→          parts.push(addr.road)
   177→        }
   178→      }
   179→      if (addr.neighbourhood) parts.push(addr.neighbourhood)
   180→      else if (addr.suburb) parts.push(addr.suburb)
   181→      if (addr.village) {
   182→        parts.push(addr.village)
   183→      } else if (addr.town) {
   184→        parts.push(addr.town)
   185→      } else if (addr.city) {
   186→        parts.push(addr.city)
   187→      }
   188→      if (addr.city && !parts.includes(addr.city)) {
   189→        parts.push(addr.city)
   190→      } else if (addr.town && !parts.includes(addr.town)) {
   191→        parts.push(addr.town)
   192→      } else if (addr.city_district) {
   193→        parts.push(addr.city_district)
   194→      }
   195→      if (addr.county) {
   196→        parts.push(addr.county)
   197→      }
   198→      if (addr.state) {
   199→        parts.push(addr.state)
   200→      }
   201→      if (addr.postcode) {
   202→        parts.push(addr.postcode)
   203→      }
   204→    } else {
   205→      if (addr.road) {
   206→        if (addr.house_number) {
   207→          parts.push(`${addr.road} ${addr.house_number}`)
   208→        } else {
   209→          parts.push(addr.road)
   210→        }
   211→      }
   212→      if (addr.neighbourhood) parts.push(addr.neighbourhood)
   213→      else if (addr.suburb) parts.push(addr.suburb)
   214→      if (addr.village) {
   215→        parts.push(addr.village)
   216→      } else if (addr.town) {
   217→        parts.push(addr.town)
   218→      } else if (addr.city) {
   219→        parts.push(addr.city)
   220→      }
   221→    }
   222→
   223→    return parts.join(', ')
   224→  }

   248→  const getCurrentLocation = async () => {

   319→  // Función para obtener ruta de OSRM (Open Source Routing Machine)
   320→  const getOSRMRoute = async (start: [number, number], end: [number, number], profile: 'walking' | 'driving' = 'driving') => {
   321→    try {
   322→      const url = `https://router.project-osrm.org/route/v1/${profile}/${start[1]},${start[0]};${end[1]},${end[0]}?overview=full&geometries=geojson`
   323→      const response = await fetch(url)
   324→      const data = await response.json()
   325→
   326→      if (data.code === 'Ok' && data.routes && data.routes.length > 0) {
   327→        const route = data.routes[0]
   328→        // Extraer las coordenadas de la ruta
   329→        const coordinates = route.geometry.coordinates.map((coord: number[]) => [coord[1], coord[0]]) as [number, number][]
   330→        return coordinates
   331→      }
   332→      return null
   333→    } catch (error) {
   334→      console.error('Error al obtener ruta de OSRM:', error)
   335→      return null
   336→    }
   337→  }

   339→  // Función para obtener paradas de autobús de OpenStreetMap usando Overpass API
   340→  const getBusStopsFromOSM = async (bounds: { south: number; west: number; north: number; east: number }) => {
   341→    try {
   342→      setLoadingBusStops(true)
   343→      const query = `
   344→        [out:json][timeout:25];
   345→        (
   346→          node["public_transport"="platform"]["bus"="yes"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});
   347→          node["highway"="bus_stop"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});
   348→        );
   349→        out body;
   350→        >;
   351→        out skel qt;
   352→      `
   353→
   354→      const response = await fetch('https://overpass-api.de/api/interpreter', {
   355→        method: 'POST',
   356→        body: query,
   357→      })
   358→
   359→      const data = await response.json()
   360→
   361→      if (data.elements && data.elements.length > 0) {
   362→        const stops: BusStop[] = data.elements
   363→          .filter((el: any) => el.type === 'node' && el.lat && el.lon)
   364→          .map((el: any) => ({
   365→            id: el.id.toString(),
   366→            name: el.tags?.name || el.tags?.ref || `Parada ${el.id}`,
   367→            lat: el.lat,
   368→            lon: el.lon,
   369→          }))
   370→
   371→        // Limitar a las 50 paradas más cercanas para no saturar el mapa
   372→        return stops.slice(0, 50)
   373→      }
   374→
   375→      return []
   376→    } catch (error) {
   377→      console.error('Error al obtener paradas de autobús:', error)
   378→      return []
   379→    } finally {
   380→      setLoadingBusStops(false)
   381→    }
   382→  }

   384→  // Calcular los límites del área alrededor de la ruta
   385→  const getRouteBounds = (routePath: RoutePath | null) => {
   386→    if (!routePath) return null
   387→
   388→    const allPoints: [number, number][] = []
   389→
   390→    if (routePath.walking) {
   391→      allPoints.push(...routePath.walking)
   392→    }
   393→    if (routePath.bus) {
   394→      allPoints.push(...routePath.bus)
   395→    }
   396→    if (routePath.direct) {
   397→      allPoints.push(...routePath.direct)
   398→    }
   399→
   400→    if (allPoints.length === 0) return null
   401→
   402→    const lats = allPoints.map(p => p[0])
   403→    const lons = allPoints.map(p => p[1])
   404→
   405→    const margin = 0.01 // Aproximadamente 1 km de margen
   406→
   407→    return {
   408→      south: Math.min(...lats) - margin,
   409→      west: Math.min(...lons) - margin,
   410→      north: Math.max(...lats) + margin,
   411→      east: Math.max(...lons) + margin,
   412→    }
   413→  }

   415→  // Calcular centro y zoom para mostrar toda la ruta
   416→  const fitRouteToBounds = (routePath: RoutePath | null) => {

   491→  const handlePlanRoute = async () => {
   492→    if (!destination.trim() || !currentLocation) {
   493→      setError('Por favor ingresa un destino y espera a obtener tu ubicación')
   494→      return
   495→    }
   496→
   497→    setPlanning(true)
   498→    setLoadingRoute(true)
   499→    setError(null)
   500→    setSelectedRoute(null)
   501→    setRoutePath(null)
   502→    setBusStops(null)
   503→
   504→    try {
   505→      const response = await fetch(
   506→        `/api/routes/plan?lat=${currentLocation.latitude}&lon=${currentLocation.longitude}&destination=${encodeURIComponent(destination)}`
   507→      )
   508→      const data = await response.json()
   509→
   510→      if (data.success) {
   511→        setPlannedRoutes(data.routes)
   512→        setNearestStop(data.nearestStop)
   513→        setHasPlanned(true)
   514→

   517→
   518→          // Obtener rutas reales de las calles usando OSRM
   519→          const route = data.routes[0]
   520→          const newRoutePath: RoutePath = {}
   521→
   522→          // Ruta caminando desde ubicación actual hasta parada de embarque
   523→          if (route.boardingStop?.coordinates && currentLocation) {
   524→            const walkingRoute = await getOSRMRoute(
   525→              [currentLocation.latitude, currentLocation.longitude],
   526→              [route.boardingStop.coordinates.latitude, route.boardingStop.coordinates.longitude],
   527→              'walking'
   528→            )
   529→            if (walkingRoute) {
   530→              newRoutePath.walking = walkingRoute
   531→            }
   532→          }
   533→
   534→          // Ruta en autobús desde parada de embarque hasta destino
   535→          if (route.boardingStop?.coordinates && route.destinationStop?.coordinates) {
   536→            const busRoute = await getOSRMRoute(
   537→              [route.boardingStop.coordinates.latitude, route.boardingStop.coordinates.longitude],
   538→              [route.destinationStop.coordinates.latitude, route.destinationStop.coordinates.longitude],
   539→              'driving'
   540→            )
   541→            if (busRoute) {
   542→              newRoutePath.bus = busRoute
   543→            }
   544→          }
   545→
   546→          setRoutePath(newRoutePath)
   547→
   548→          // Ajustar mapa para mostrar toda la ruta
   549→          fitRouteToBounds(newRoutePath)
   550→
   551→          // Obtener paradas de autobús de OpenStreetMap alrededor de la ruta
   552→          const bounds = getRouteBounds(newRoutePath)
   553→          if (bounds) {
   554→            const stops = await getBusStopsFromOSM(bounds)
   555→            setBusStops(stops)
   556→          }
   557→        }
   558→
   559→        if (data.routes.length === 0) {
   560→          setError('No se encontraron rutas hacia ese destino. Intenta con otro destino.')
   561→        }
   562→      } else {
   563→        setError(data.error || 'Error al planificar la ruta')
   564→      }
   565→    } catch (error) {
   566→      console.error('Error al planificar ruta:', error)
   567→      setError('Error de conexión. Por favor intenta de nuevo.')
   568→    } finally {
   569→      setPlanning(false)
   570→      setLoadingRoute(false)
   571→    }
   572→  }

   579→    }
   580→
   581→    setIsTracking(false)
   582→    setTripStartTime(null)
   583→    setElapsedTime(0)
   584→    setDistanceRemaining(0)
   585→    setShowArrivalNotification(false)
   586→    setTrackingPanelVisible(true)
   587→    setShowStartTripAlert(false)
   588→
   589→    setDestination('')
   590→    setSelectedDestination(null)
   591→    setPlannedRoutes([])

   631→    const dLon = (lon2 - lon1) * Math.PI / 180
   632→    const a =
   633→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   634→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   635→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   636→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   637→    return R * c
   638→  }
