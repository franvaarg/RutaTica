   344→  // Función para obtener ruta de OSRM (Open Source Routing Machine)

   346→    try {
   347→      const url = `https://router.project-osrm.org/route/v1/${profile}/${start[1]},${start[0]};${end[1]},${end[0]}?overview=full&geometries=geojson`
   348→      const response = await fetch(url)
   349→      const data = await response.json()
   350→
   351→      if (data.code === 'Ok' && data.routes && data.routes.length > 0) {
   352→        const route = data.routes[0]
   353→        // Extraer las coordenadas de la ruta
   354→        const coordinates = route.geometry.coordinates.map((coord: number[]) => [coord[1], coord[0]]) as [number, number][]
   355→        return coordinates

   357→      return null

   409→  // Calcular los límites del área alrededor de la ruta
   410→  const getRouteBounds = (routePath: RoutePath | null) => {
   411→    if (!routePath) return null
   412→
   413→    const allPoints: [number, number][] = []
   414→
   415→    if (routePath.walking) {
   416→      allPoints.push(...routePath.walking)
   417→    }
   418→    if (routePath.bus) {
   419→      allPoints.push(...routePath.bus)
   420→    }
   421→    if (routePath.direct) {
   422→      allPoints.push(...routePath.direct)
   423→    }
   424→
   425→    if (allPoints.length === 0) return null
   426→
   427→    const lats = allPoints.map(p => p[0])
   428→    const lons = allPoints.map(p => p[1])
   429→
   430→    const margin = 0.01 // Aproximadamente 1 km de margen
   431→
   432→    return {
   433→      south: Math.min(...lats) - margin,
   434→      west: Math.min(...lons) - margin,
   435→      north: Math.max(...lats) + margin,
   436→      east: Math.max(...lons) + margin,
   437→    }
   438→  }

   719→    const dLat = (lat2 - lat1) * Math.PI / 180
   720→    const dLon = (lon2 - lon1) * Math.PI / 180
   721→    const a =
   722→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   723→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   724→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   725→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   726→    return R * c
