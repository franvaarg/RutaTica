   145→  const getAddressFromCoordinates = async (lat: number, lon: number): Promise<string> => {

   159→  const formatAddress = (data: any, useFullAddress: boolean): string => {
   160→    if (!data || !data.address) return ''
   161→    const addr = data.address
   162→    const parts: string[] = []
   163→
   164→    if (useFullAddress) {
   165→      if (addr.road) {
   166→        if (addr.house_number) {
   167→          parts.push(`${addr.road} ${addr.house_number}`)
   168→        } else {
   169→          parts.push(addr.road)
   170→        }
   171→      }
   172→      if (addr.neighbourhood) parts.push(addr.neighbourhood)
   173→      else if (addr.suburb) parts.push(addr.suburb)
   174→      if (addr.village) {
   175→        parts.push(addr.village)
   176→      } else if (addr.town) {
   177→        parts.push(addr.town)
   178→      } else if (addr.city) {
   179→        parts.push(addr.city)
   180→      }
   181→      if (addr.city && !parts.includes(addr.city)) {
   182→        parts.push(addr.city)
   183→      } else if (addr.town && !parts.includes(addr.town)) {
   184→        parts.push(addr.town)
   185→      } else if (addr.city_district) {
   186→        parts.push(addr.city_district)
   187→      }
   188→      if (addr.county) {
   189→        parts.push(addr.county)
   190→      }
   191→      if (addr.state) {
   192→        parts.push(addr.state)
   193→      }
   194→      if (addr.postcode) {
   195→        parts.push(addr.postcode)
   196→      }
   197→    } else {
   198→      if (addr.road) {
   199→        if (addr.house_number) {
   200→          parts.push(`${addr.road} ${addr.house_number}`)
   201→        } else {
   202→          parts.push(addr.road)
   203→        }
   204→      }
   205→      if (addr.neighbourhood) parts.push(addr.neighbourhood)
   206→      else if (addr.suburb) parts.push(addr.suburb)
   207→      if (addr.village) {
   208→        parts.push(addr.village)
   209→      } else if (addr.town) {
   210→        parts.push(addr.town)
   211→      } else if (addr.city) {
   212→        parts.push(addr.city)
   213→      }
   214→    }
   215→
   216→    return parts.join(', ')
   217→  }

   303→  // Función para obtener ruta de OSRM (Open Source Routing Machine)
   304→  const getOSRMRoute = async (start: [number, number], end: [number, number], profile: 'walking' | 'driving' = 'driving') => {
   305→    try {
   306→      const url = `https://router.project-osrm.org/route/v1/${profile}/${start[1]},${start[0]};${end[1]},${end[0]}?overview=full&geometries=geojson`
   307→      const response = await fetch(url)
   308→      const data = await response.json()
   309→
   310→      if (data.code === 'Ok' && data.routes && data.routes.length > 0) {
   311→        const route = data.routes[0]
   312→        // Extraer las coordenadas de la ruta
   313→        const coordinates = route.geometry.coordinates.map((coord: number[]) => [coord[1], coord[0]]) as [number, number][]
   314→        return coordinates
   315→      }
   316→      return null
   317→    } catch (error) {
   318→      console.error('Error al obtener ruta de OSRM:', error)
   319→      return null
   320→    }
   321→  }

   323→  // Función para obtener paradas de autobús de OpenStreetMap usando Overpass API
   324→  const getBusStopsFromOSM = async (bounds: { south: number; west: number; north: number; east: number }) => {
   325→    try {
   326→      setLoadingBusStops(true)
   327→      const query = `
   328→        [out:json][timeout:25];
   329→        (
   330→          node["public_transport"="platform"]["bus"="yes"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});
   331→          node["highway"="bus_stop"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});
   332→        );
   333→        out body;
   334→        >;
   335→        out skel qt;
   336→      `
   337→
   338→      const response = await fetch('https://overpass-api.de/api/interpreter', {
   339→        method: 'POST',
   340→        body: query,
   341→      })
   342→
   343→      const data = await response.json()
   344→
   345→      if (data.elements && data.elements.length > 0) {
   346→        const stops: BusStop[] = data.elements
   347→          .filter((el: any) => el.type === 'node' && el.lat && el.lon)
   348→          .map((el: any) => ({
   349→            id: el.id.toString(),
   350→            name: el.tags?.name || el.tags?.ref || `Parada ${el.id}`,
   351→            lat: el.lat,
   352→            lon: el.lon,
   353→          }))
   354→
   355→        // Limitar a las 50 paradas más cercanas para no saturar el mapa
   356→        return stops.slice(0, 50)
   357→      }
   358→
   359→      return []
   360→    } catch (error) {
   361→      console.error('Error al obtener paradas de autobús:', error)
   362→      return []
   363→    } finally {
   364→      setLoadingBusStops(false)
   365→    }
   366→  }

   368→  // Calcular los límites del área alrededor de la ruta
   369→  const getRouteBounds = (routePath: RoutePath | null) => {
   370→    if (!routePath) return null
   371→
   372→    const allPoints: [number, number][] = []
   373→
   374→    if (routePath.walking) {
   375→      allPoints.push(...routePath.walking)
   376→    }
   377→    if (routePath.bus) {
   378→      allPoints.push(...routePath.bus)
   379→    }
   380→    if (routePath.direct) {
   381→      allPoints.push(...routePath.direct)
   382→    }
   383→
   384→    if (allPoints.length === 0) return null
   385→
   386→    const lats = allPoints.map(p => p[0])
   387→    const lons = allPoints.map(p => p[1])
   388→
   389→    const margin = 0.01 // Aproximadamente 1 km de margen
   390→
   391→    return {
   392→      south: Math.min(...lats) - margin,
   393→      west: Math.min(...lons) - margin,
   394→      north: Math.max(...lats) + margin,
   395→      east: Math.max(...lons) + margin,
   396→    }
   397→  }

   399→  // Calcular centro y zoom para mostrar toda la ruta
   400→  const fitRouteToBounds = (routePath: RoutePath | null) => {
   401→    if (!routePath) return
   402→
   403→    const bounds = getRouteBounds(routePath)
   404→    if (!bounds || !currentLocation) return
   405→
   406→    // Validar que todos los valores del bounds sean finitos
   407→    const { south, west, north, east } = bounds
   408→    if (!isFinite(south) || !isFinite(west) || !isFinite(north) || !isFinite(east)) {
   409→      console.error('Bounds calculado contiene valores inválidos:', bounds)
   410→      return
   411→    }
   412→
   413→    // Validar que el bounds tenga sentido geográfico
   414→    if (south >= north || west >= east) {
   415→      console.error('Bounds inválido: south debe ser menor que north, west menor que east:', bounds)
   416→      return
   417→    }
   418→
   419→    const centerLat = (north + south) / 2
   420→    const centerLon = (east + west) / 2
   421→
   422→    setMapCenter([centerLat, centerLon])
   423→    setMapZoom(10) // Zoom out para mostrar toda la ruta

   478→  const handlePlanRoute = async () => {
   479→    if (!destination.trim() || !currentLocation) {
   480→      setError('Por favor ingresa un destino y espera a obtener tu ubicación')
   481→      return
   482→    }
   483→
   484→    setPlanning(true)
   485→    setLoadingRoute(true)
   486→    setError(null)
   487→    setSelectedRoute(null)
   488→    setRoutePath(null)
   489→    setBusStops(null)
   490→
   491→    try {
   492→      const response = await fetch(
   493→        `/api/routes/plan?lat=${currentLocation.latitude}&lon=${currentLocation.longitude}&destination=${encodeURIComponent(destination)}`
   494→      )
   495→      const data = await response.json()
   496→
   497→      if (data.success) {
   498→        setPlannedRoutes(data.routes)
   499→        setNearestStop(data.nearestStop)
   500→        setHasPlanned(true)
   501→
   502→        if (data.routes.length > 0) {
   503→          setSelectedRoute(data.routes[0])
   504→
   505→          // Obtener rutas reales de las calles usando OSRM
   506→          const route = data.routes[0]
   507→          const newRoutePath: RoutePath = {}
   508→
   509→          // Ruta caminando desde ubicación actual hasta parada de embarque
   510→          if (route.boardingStop?.coordinates && currentLocation) {
   511→            const walkingRoute = await getOSRMRoute(
   512→              [currentLocation.latitude, currentLocation.longitude],
   513→              [route.boardingStop.coordinates.latitude, route.boardingStop.coordinates.longitude],
   514→              'walking'
   515→            )
   516→            if (walkingRoute) {
   517→              newRoutePath.walking = walkingRoute
   518→            }
   519→          }
   520→
   521→          // Ruta en autobús desde parada de embarque hasta destino
   522→          if (route.boardingStop?.coordinates && route.destinationStop?.coordinates) {
   523→            const busRoute = await getOSRMRoute(
   524→              [route.boardingStop.coordinates.latitude, route.boardingStop.coordinates.longitude],
   525→              [route.destinationStop.coordinates.latitude, route.destinationStop.coordinates.longitude],
   526→              'driving'
   527→            )
   528→            if (busRoute) {
   529→              newRoutePath.bus = busRoute
   530→            }
   531→          }
   532→
   533→          setRoutePath(newRoutePath)
   534→
   535→          // Ajustar mapa para mostrar toda la ruta
   536→          fitRouteToBounds(newRoutePath)
   537→
   538→          // Obtener paradas de autobús de OpenStreetMap alrededor de la ruta
   539→          const bounds = getRouteBounds(newRoutePath)
   540→          if (bounds) {
   541→            const stops = await getBusStopsFromOSM(bounds)
   542→            setBusStops(stops)
   543→          }
   544→        }
   545→
   546→        if (data.routes.length === 0) {
   547→          setError('No se encontraron rutas hacia ese destino. Intenta con otro destino.')
   548→        }
   549→      } else {
   550→        setError(data.error || 'Error al planificar la ruta')
   551→      }
   552→    } catch (error) {
   553→      console.error('Error al planificar ruta:', error)
   554→      setError('Error de conexión. Por favor intenta de nuevo.')
   555→    } finally {
   556→      setPlanning(false)
   557→      setLoadingRoute(false)
   558→    }
   559→  }

   571→    setDistanceRemaining(0)
   572→    setShowArrivalNotification(false)
   573→    setTrackingPanelVisible(true)
   574→    setShowStartTripDialog(false)

   578→    setPlannedRoutes([])
   579→    setSelectedRoute(null)

   589→    }

   617→    const dLat = (lat2 - lat1) * Math.PI / 180

   620→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   621→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   622→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   623→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   624→    return R * c
   625→  }
