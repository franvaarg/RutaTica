   188→        parts.push(addr.city)
   189→      }
   190→      if (addr.city && !parts.includes(addr.city)) {
   191→        parts.push(addr.city)
   192→      } else if (addr.town && !parts.includes(addr.town)) {
   193→        parts.push(addr.town)
   194→      } else if (addr.city_district) {
   195→        parts.push(addr.city_district)
   196→      }
   197→      if (addr.county) {
   198→        parts.push(addr.county)
   199→      }
   200→      if (addr.state) {
   201→        parts.push(addr.state)
   202→      }
   203→      if (addr.postcode) {

   321→  // Función para obtener ruta de OSRM (Open Source Routing Machine)
   322→  const getOSRMRoute = async (start: [number, number], end: [number, number], profile: 'walking' | 'driving' = 'driving') => {
   323→    try {
   324→      const url = `https://router.project-osrm.org/route/v1/${profile}/${start[1]},${start[0]};${end[1]},${end[0]}?overview=full&geometries=geojson`
   325→      const response = await fetch(url)
   326→      const data = await response.json()
   327→
   328→      if (data.code === 'Ok' && data.routes && data.routes.length > 0) {
   329→        const route = data.routes[0]
   330→        // Extraer las coordenadas de la ruta
   331→        const coordinates = route.geometry.coordinates.map((coord: number[]) => [coord[1], coord[0]]) as [number, number][]
   332→        return coordinates
   333→      }
   334→      return null
   335→    } catch (error) {
   336→      console.error('Error al obtener ruta de OSRM:', error)

   338→    }

   341→  // Función para obtener paradas de autobús de OpenStreetMap usando Overpass API
   342→  const getBusStopsFromOSM = async (bounds: { south: number; west: number; north: number; east: number }) => {
   343→    try {
   344→      setLoadingBusStops(true)
   345→      const query = `
   346→        [out:json][timeout:25];
   347→        (
   348→          node["public_transport"="platform"]["bus"="yes"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});
   349→          node["highway"="bus_stop"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});
   350→        );
   351→        out body;
   352→        >;
   353→        out skel qt;
   354→      `
   355→
   356→      const response = await fetch('https://overpass-api.de/api/interpreter', {
   357→        method: 'POST',
   358→        body: query,
   359→      })
   360→
   361→      const data = await response.json()
   362→
   363→      if (data.elements && data.elements.length > 0) {
   364→        const stops: BusStop[] = data.elements
   365→          .filter((el: any) => el.type === 'node' && el.lat && el.lon)
   366→          .map((el: any) => ({
   367→            id: el.id.toString(),
   368→            name: el.tags?.name || el.tags?.ref || `Parada ${el.id}`,
   369→            lat: el.lat,
   370→            lon: el.lon,
   371→          }))
   372→
   373→        // Limitar a las 50 paradas más cercanas para no saturar el mapa
   374→        return stops.slice(0, 50)
   375→      }
   376→
   377→      return []
   378→    } catch (error) {
   379→      console.error('Error al obtener paradas de autobús:', error)
   380→      return []
   381→    } finally {
   382→      setLoadingBusStops(false)
   383→    }
   384→  }

   386→  // Calcular los límites del área alrededor de la ruta
   387→  const getRouteBounds = (routePath: RoutePath | null) => {
   388→    if (!routePath) return null
   389→
   390→    const allPoints: [number, number][] = []
   391→
   392→    if (routePath.walking) {
   393→      allPoints.push(...routePath.walking)
   394→    }
   395→    if (routePath.bus) {
   396→      allPoints.push(...routePath.bus)
   397→    }
   398→    if (routePath.direct) {
   399→      allPoints.push(...routePath.direct)
   400→    }
   401→
   402→    if (allPoints.length === 0) return null
   403→
   404→    // Validar que todos los puntos tengan coordenadas válidas
   405→    const validPoints = allPoints.filter(p =>
   406→      isFinite(p[0]) && isFinite(p[1])
   407→    )
   408→
   409→    if (validPoints.length === 0) {
   410→      console.warn('No hay puntos válidos para calcular bounds', allPoints)
   411→      return null
   412→    }
   413→
   414→    const lats = validPoints.map(p => p[0])
   415→    const lons = validPoints.map(p => p[1])
   416→
   417→    const margin = 0.01 // Aproximadamente 1 km de margen
   418→
   419→    return {
   420→      south: Math.min(...lats) - margin,
   421→      west: Math.min(...lons) - margin,
   422→      north: Math.max(...lats) + margin,
   423→      east: Math.max(...lons) + margin,
   424→    }
   425→  }

   545→            }
   546→          }
   547→
   548→          // Ruta en autobús desde parada de embarque hasta destino
   549→          if (route.boardingStop?.coordinates && route.destinationStop?.coordinates) {
   550→            const busRoute = await getOSRMRoute(
   551→              [route.boardingStop.coordinates.latitude, route.boardingStop.coordinates.longitude],
   552→              [route.destinationStop.coordinates.latitude, route.destinationStop.coordinates.longitude],
   553→              'driving'
   554→            )
   555→            if (busRoute) {
   556→              newRoutePath.bus = busRoute
   557→            }
   558→          }
   559→
   560→          setRoutePath(newRoutePath)
   561→
   562→          // Ajustar mapa para mostrar toda la ruta
   563→          fitRouteToBounds(newRoutePath)
   564→
   565→          // Obtener paradas de autobús de OpenStreetMap alrededor de la ruta
   566→          const bounds = getRouteBounds(newRoutePath)
   567→          if (bounds) {
   568→            const stops = await getBusStopsFromOSM(bounds)
   569→            setBusStops(stops)
   570→          }
   571→        }
   572→
   573→        if (data.routes.length === 0) {
   574→          setError('No se encontraron rutas hacia ese destino. Intenta con otro destino.')
   575→        }
   576→      } else {
   577→        setError(data.error || 'Error al planificar la ruta')
   578→      }
   579→    } catch (error) {
   580→      console.error('Error al planificar ruta:', error)
   581→      setError('Error de conexión. Por favor intenta de nuevo.')
   582→    } finally {
   583→      setPlanning(false)
   584→      setLoadingRoute(false)
   585→    }
   586→  }

   645→    const dLon = (lon2 - lon1) * Math.PI / 180

   647→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   648→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   649→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   650→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   651→    return R * c
   652→  }
