    11→// Función para obtener Leaflet de forma segura (optimizada con cache de promesa)
    12→async function getLeaflet() {
    13→  if (typeof window === 'undefined') return null
    14→
    15→  // Usar cache de promesa para evitar múltiples llamadas simultáneas
    16→  if (!leafletPromise) {
    17→    leafletPromise = import('leaflet').then((L) => {
    18→      // Fix para iconos por defecto de Leaflet en Next.js
    19→      delete (L.Icon.Default.prototype as any)._getIconUrl
    20→      L.Icon.Default.mergeOptions({
    21→        iconRetinaUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png',
    22→        iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png',
    23→        shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png',
    24→      })
    25→      leafletInstance = L
    26→      return L
    27→    })
    28→  }
    29→
    30→  return leafletPromise
    31→}

   549→          weight: 6,
   550→        })
   551→      }
   552→    }
   553→    // Si no hay ruta seleccionada pero hay destino seleccionado con camino directo
   554→    else if (routePath?.direct && routePath.direct.length > 0) {
   555→      polylines.push({
   556→        positions: routePath.direct,
   557→        color: '#E31837', // Rojo para conexión directa
   558→        weight: 4,
   559→        dashArray: '5, 5',
   560→      })
   561→    }
   562→    // Fallback: Línea directa simple
   563→    else if (userLocation && destinationCoordinates) {
   564→      polylines.push({
   565→        positions: [
   566→          userLocation,
   567→          [destinationCoordinates.latitude, destinationCoordinates.longitude]
   568→        ],
   569→        color: '#E31837',
   570→        weight: 4,
   571→        dashArray: '5, 5',
   572→      })
   573→    }
   574→
   575→    return polylines
   576→  }
   577→
   578→  // No renderizar nada hasta que el componente esté montado en el cliente
   579→  if (!isMounted || !L || !userIcon || !originIcon || !stopIcon || !boardingIcon || !destinationIcon || !busStationIcon) {
   580→    return (
   581→      <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden border border-[#E5E7EB] flex items-center justify-center bg-gray-50">
   582→        <div className="text-center">
   583→          <div className="w-12 h-12 border-4 border-[#E31837] border-t-transparent rounded-full animate-spin mx-auto mb-3"></div>
   584→          <p className="text-sm text-[#6B7280]">Cargando mapa...</p>
   585→        </div>
   586→      </div>
   587→    )
   588→  }
   589→
   590→  return (
   591→    <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden">
   592→      <MapContainer
   593→        center={center}
   594→        zoom={zoom}
   595→        className="w-full h-full"
   596→        scrollWheelZoom={false}
   597→        zoomControl={false}
   598→        ref={mapRef}
   599→      >
   600→        {/* Capa de OpenStreetMap */}
   601→        <TileLayer
   602→          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
   603→          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
   604→        />
   605→
   606→        {/* Controles de zoom personalizados */}
   607→        <ZoomControls />
   608→
   609→        {/* Polilíneas de la ruta seleccionada o conexión directa */}
   610→        {getRoutePolylines().map((polyline, index) => (
   611→          <Polyline
   612→            key={index}
   613→            positions={polyline.positions}
   614→            pathOptions={{
   615→              color: polyline.color,
   616→              weight: polyline.weight,
   617→              dashArray: polyline.dashArray,
   618→              opacity: 0.8,
   619→            }}
   620→          />
   621→        ))}
   622→
   623→        {/* Marcador de origen (ubicación del usuario) */}
   624→        {userLocation && (
   625→          <Marker
   626→            position={userLocation}
   627→            icon={isTracking ? trackingIcon : (selectedRoute ? userIcon : originIcon)}
   628→          >
   629→            <Popup>
   630→              <div className="text-sm p-1 min-w-32">
   631→                <strong className={isTracking ? "text-[#E31837]" : "text-[#0052B4]"}>
   632→                  {isTracking ? "🚗 Tu posición" : "📍 Origen"}
   633→                </strong>
   634→                <br />
   635→                <span className="text-xs text-[#6B7280]">
   636→                  {isTracking ? "Rastreando en tiempo real" : "Tu ubicación actual"}
   637→                </span>
   638→              </div>
   639→            </Popup>
   640→          </Marker>
   641→        )}
   642→
   643→        {/* Marcadores de paradas de buses de la base de datos (resaltados) */}
   644→        {dbStops && dbStops.length > 0 && (
   645→          <>
   646→            {dbStops.map((stop) => (
   647→              <Marker
   648→                key={`db-${stop.id}`}
   649→                position={[stop.latitude, stop.longitude]}
   650→                icon={busStationIcon}
   651→              >
   652→                <Popup>
   653→                  <div className="text-sm p-1 min-w-40">
   654→                    <div className="flex items-center gap-2 mb-2">
   655→                      <div className="w-6 h-6 bg-[#E31837] rounded-full flex items-center justify-center">
   656→                        <svg viewBox="0 0 24 24" className="w-4 h-4 text-white" fill="currentColor">
   657→                          <path d="M4 16c0 .88.39 1.67 1 2.22V20c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1h8v1c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1.78c.61-.55 1-1.34 1-2.22V6c0-3.5-3.58-4-8-4s-8 .5-8 4v10zm3.5 1c-.83 0-1.5-.67-1.5-1.5S6.67 14 7.5 14s1.5.67 1.5 1.5S8.33 17 7.5 17zm9 0c-.83 0-1.5-.67-1.5-1.5s.67-1.5 1.5-1.5 1.5.67 1.5 1.5-.67 1.5-1.5 1.5zm1.5-6H6V6h12v5z"/>
   658→                        </svg>
   659→                      </div>
   660→                      <strong className="text-[#E31837]">Parada de Bus</strong>
   661→                    </div>
   662→                    <p className="font-semibold text-[#374151]">{stop.name}</p>
   663→                    {stop.city && <p className="text-xs text-[#6B7280] mt-1">{stop.city}</p>}
   664→                    {stop.distance !== undefined && (
   665→                      <p className="text-xs text-[#0052B4] mt-1 font-medium">
   666→                        {stop.distance.toFixed(1)} km de tu ubicación
   667→                      </p>
   668→                    )}
   669→                  </div>
   670→                </Popup>
   671→              </Marker>
   672→            ))}
   673→          </>
   674→        )}
   675→
   676→        {/* Marcador de parada de embarque */}
   677→        {selectedRoute?.boardingStop?.coordinates && (
   678→          <Marker
   679→            position={[selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude]}
   680→            icon={boardingIcon}
   681→          >
   682→            <Popup>
   683→              <div className="text-sm p-1 min-w-32">
   684→                <strong className="text-orange-600">⬆️ Sube aquí</strong>
   685→                <br />
   686→                {selectedRoute.boardingStop.name}
   687→                {selectedRoute.boardingStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.boardingStop.city}</span></>}
   688→              </div>
   689→            </Popup>
   690→          </Marker>
   691→        )}
   692→
   693→        {/* Marcador de destino */}
   694→        {selectedRoute?.destinationStop?.coordinates && (
   695→          <Marker
   696→            position={[selectedRoute.destinationStop.coordinates.latitude, selectedRoute.destinationStop.coordinates.longitude]}
   697→            icon={destinationIcon}
   698→          >
   699→            <Popup>
   700→              <div className="text-sm p-1 min-w-32">
   701→                <strong className="text-[#E31837]">🏁 Destino</strong>
   702→                <br />
   703→                {selectedRoute.destinationStop.name}
   704→                {selectedRoute.destinationStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.destinationStop.city}</span></>}
   705→              </div>
   706→            </Popup>
   707→          </Marker>
   708→        )}
   709→
   710→        {/* Marcador de destino seleccionado */}
   711→        {destinationCoordinates && (
   712→          <Marker
   713→            position={[destinationCoordinates.latitude, destinationCoordinates.longitude]}
   714→            icon={destinationIcon}
   715→          >
   716→            <Popup>
   717→              <div className="text-sm p-1 min-w-32">
   718→                <strong className="text-[#E31837]">🏁 Destino</strong>
   719→                <br />
   720→                {destinationCoordinates.displayName || destinationCoordinates.name}
   721→              </div>
   722→            </Popup>
   723→          </Marker>
   724→        )}
   725→
   726→        {/* Marcador de parada más cercana (solo cuando no hay ruta ni destino seleccionado) */}
   727→        {!selectedRoute && !destinationCoordinates && nearestStop && (
   728→          <Marker
   729→            position={[nearestStop.coordinates.latitude, nearestStop.coordinates.longitude]}
   730→            icon={stopIcon}
   731→          >
   732→            <Popup>
   733→              <div className="text-sm p-1 min-w-32">
   734→                <strong className="text-[#10B981]">🚌 Parada más cercana</strong>
   735→                <br />
   736→                {nearestStop.name}
   737→              </div>
   738→            </Popup>
   739→          </Marker>
   740→        )}
   741→
   742→        {/* Marcadores de paradas de autobús de OpenStreetMap */}
   743→        {busStops && busStops.length > 0 && (
   744→          <>
   745→            {busStops.map((stop) => (
   746→              <Marker
   747→                key={stop.id}
   748→                position={[stop.lat, stop.lon]}
   749→                icon={stopIcon}
   750→              >
   751→                <Popup>
   752→                  <div className="text-sm p-1 min-w-32">
   753→                    <strong className="text-[#10B981]">🚌 Parada</strong>
   754→                    <br />
   755→                    {stop.name || 'Sin nombre'}
   756→                  </div>
   757→                </Popup>
   758→              </Marker>
   759→            ))}
   760→          </>
   761→        )}
   762→      </MapContainer>
   763→    </div>
   764→  )
   765→}
   766→
   767→export default BusMap
