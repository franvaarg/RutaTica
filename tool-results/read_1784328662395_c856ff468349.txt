    11→// Función para obtener Leaflet de forma segura (optimizada con cache de promesa)
    12→async function getLeaflet() {
    13→  if (typeof window === 'undefined') return null
    14→
    15→  // Usar cache de promesa para evitar múltiples llamadas simultáneas
    16→  if (!leafletPromise) {
    17→    leafletPromise = import('leaflet').then((L) => {
    18→      // Fix para iconos por defecto de Leaflet en Next.js
    19→      delete (L.Icon.Default.prototype as any)._getIconUrl
    20→      L.Icon.Default.mergeOptions({
    21→        iconRetinaUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png',
    22→        iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png',
    23→        shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png',
    24→      })
    25→      leafletInstance = L
    26→      return L
    27→    })
    28→  }
    29→
    30→  return leafletPromise
    31→}

   540→      // Segmento 2: Ruta de autobús usando GTFS shape data o ruta calculada
   541→      if (routePath.bus && routePath.bus.length > 0) {
   542→        polylines.push({
   543→          positions: routePath.bus,
   544→          color: '#16a34a', // Verde para autobús
   545→          weight: 6,
   546→        })
   547→      }
   548→
   549→      // Segmento 3: Camino a pie desde parada de bajada hasta destino final
   550→      if (routePath.walking2 && routePath.walking2.length > 0) {
   551→        polylines.push({
   552→          positions: routePath.walking2,
   553→          color: '#0052B4', // Azul para caminar
   554→          weight: 5,
   555→          dashArray: '10, 10',
   556→        })
   557→      }
   558→    }
   559→    // Fallback: Si no hay ruta con caminos reales pero hay ruta seleccionada, usar líneas rectas
   560→    else if (selectedRoute && userLocation) {
   561→      if (selectedRoute.boardingStop?.coordinates) {
   562→        polylines.push({
   563→          positions: [
   564→            userLocation,
   565→            [selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude]
   566→          ],
   567→          color: '#0052B4',
   568→          weight: 5,
   569→          dashArray: '10, 10',
   570→        })
   571→      }
   572→
   573→      if (selectedRoute.boardingStop?.coordinates && selectedRoute.destinationStop?.coordinates) {
   574→        polylines.push({
   575→          positions: [
   576→            [selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude],
   577→            [selectedRoute.destinationStop.coordinates.latitude, selectedRoute.destinationStop.coordinates.longitude]
   578→          ],
   579→          color: '#16a34a',
   580→          weight: 6,
   581→        })
   582→      }
   583→    }
   584→    // Si no hay ruta seleccionada pero hay destino seleccionado con camino directo
   585→    else if (routePath?.direct && routePath.direct.length > 0) {
   586→      polylines.push({
   587→        positions: routePath.direct,
   588→        color: '#E31837', // Rojo para conexión directa
   589→        weight: 4,
   590→        dashArray: '5, 5',
   591→      })
   592→    }
   593→    // Fallback: Línea directa simple
   594→    else if (userLocation && destinationCoordinates) {
   595→      polylines.push({
   596→        positions: [
   597→          userLocation,
   598→          [destinationCoordinates.latitude, destinationCoordinates.longitude]
   599→        ],
   600→        color: '#E31837',
   601→        weight: 4,
   602→        dashArray: '5, 5',
   603→      })
   604→    }
   605→
   606→    return polylines
   607→  }
   608→
   609→  // No renderizar nada hasta que el componente esté montado en el cliente
   610→  if (!isMounted || !L || !userIcon || !originIcon || !stopIcon || !boardingIcon || !destinationIcon || !busStationIcon) {
   611→    return (
   612→      <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden border border-[#E5E7EB] flex items-center justify-center bg-gray-50">
   613→        <div className="text-center">
   614→          <div className="w-12 h-12 border-4 border-[#E31837] border-t-transparent rounded-full animate-spin mx-auto mb-3"></div>
   615→          <p className="text-sm text-[#6B7280]">Cargando mapa...</p>
   616→        </div>
   617→      </div>
   618→    )
   619→  }
   620→
   621→  return (
   622→    <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden">
   623→      <MapContainer
   624→        center={center}
   625→        zoom={zoom}
   626→        className="w-full h-full"
   627→        scrollWheelZoom={false}
   628→        zoomControl={false}
   629→        ref={mapRef}
   630→        whenReady={() => {
   631→          // Se dispara una vez que Leaflet terminó su inicialización interna
   632→          // (incluyendo el cálculo del tamaño del contenedor). A partir de aquí
   633→          // es seguro llamar a fitBounds()/setView() sin riesgo de NaN.
   634→          setMapReady(true)
   635→          // Por si el contenedor cambia de tamaño justo después del montaje inicial
   636→          requestAnimationFrame(() => {
   637→            mapRef.current?.invalidateSize()
   638→          })
   639→        }}
   640→      >
   641→        {/* Capa de OpenStreetMap */}
   642→        <TileLayer
   643→          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
   644→          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
   645→        />
   646→
   647→        {/* Controles de zoom personalizados */}
   648→        <ZoomControls />
   649→
   650→        {/* Polilíneas de la ruta seleccionada o conexión directa */}
   651→        {getRoutePolylines().map((polyline, index) => (
   652→          <Polyline
   653→            key={index}
   654→            positions={polyline.positions}
   655→            pathOptions={{
   656→              color: polyline.color,
   657→              weight: polyline.weight,
   658→              dashArray: polyline.dashArray,
   659→              opacity: 0.8,
   660→            }}
   661→          />
   662→        ))}
   663→
   664→        {/* Marcador de origen (ubicación del usuario) */}
   665→        {userLocation && (
   666→          <Marker
   667→            position={userLocation}
   668→            icon={isTracking ? trackingIcon : (selectedRoute ? userIcon : originIcon)}
   669→          >
   670→            <Popup>
   671→              <div className="text-sm p-1 min-w-32">
   672→                <strong className={isTracking ? "text-[#E31837]" : "text-[#0052B4]"}>
   673→                  {isTracking ? "🚗 Tu posición" : "📍 Origen"}
   674→                </strong>
   675→                <br />
   676→                <span className="text-xs text-[#6B7280]">
   677→                  {isTracking ? "Rastreando en tiempo real" : "Tu ubicación actual"}
   678→                </span>
   679→              </div>
   680→            </Popup>
   681→          </Marker>
   682→        )}
   683→
   684→        {/* Marcadores de paradas de buses de la base de datos (resaltados) */}
   685→        {dbStops && dbStops.length > 0 && (
   686→          <>
   687→            {dbStops.map((stop) => (
   688→              <Marker
   689→                key={`db-${stop.id}`}
   690→                position={[stop.latitude, stop.longitude]}
   691→                icon={busStationIcon}
   692→              >
   693→                <Popup>
   694→                  <div className="text-sm p-1 min-w-40">
   695→                    <div className="flex items-center gap-2 mb-2">
   696→                      <div className="w-6 h-6 bg-[#E31837] rounded-full flex items-center justify-center">
   697→                        <svg viewBox="0 0 24 24" className="w-4 h-4 text-white" fill="currentColor">
   698→                          <path d="M4 16c0 .88.39 1.67 1 2.22V20c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1h8v1c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1.78c.61-.55 1-1.34 1-2.22V6c0-3.5-3.58-4-8-4s-8 .5-8 4v10zm3.5 1c-.83 0-1.5-.67-1.5-1.5S6.67 14 7.5 14s1.5.67 1.5 1.5S8.33 17 7.5 17zm9 0c-.83 0-1.5-.67-1.5-1.5s.67-1.5 1.5-1.5 1.5.67 1.5 1.5-.67 1.5-1.5 1.5zm1.5-6H6V6h12v5z"/>
   699→                        </svg>
   700→                      </div>
   701→                      <strong className="text-[#E31837]">Parada de Bus</strong>
   702→                    </div>
   703→                    <p className="font-semibold text-[#374151]">{stop.name}</p>
   704→                    {stop.city && <p className="text-xs text-[#6B7280] mt-1">{stop.city}</p>}
   705→                    {stop.distance !== undefined && (
   706→                      <p className="text-xs text-[#0052B4] mt-1 font-medium">
   707→                        {stop.distance.toFixed(1)} km de tu ubicación
   708→                      </p>
   709→                    )}
   710→                  </div>
   711→                </Popup>
   712→              </Marker>
   713→            ))}
   714→          </>
   715→        )}
   716→
   717→        {/* Marcador de parada de embarque */}
   718→        {selectedRoute?.boardingStop?.coordinates && (
   719→          <Marker
   720→            position={[selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude]}
   721→            icon={boardingIcon}
   722→          >
   723→            <Popup>
   724→              <div className="text-sm p-1 min-w-32">
   725→                <strong className="text-orange-600">⬆️ Sube aquí</strong>
   726→                <br />
   727→                {selectedRoute.boardingStop.name}
   728→                {selectedRoute.boardingStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.boardingStop.city}</span></>}
   729→              </div>
   730→            </Popup>
   731→          </Marker>
   732→        )}
   733→
   734→        {/* Marcador de destino */}
   735→        {selectedRoute?.destinationStop?.coordinates && (
   736→          <Marker
   737→            position={[selectedRoute.destinationStop.coordinates.latitude, selectedRoute.destinationStop.coordinates.longitude]}
   738→            icon={destinationIcon}
   739→          >
   740→            <Popup>
   741→              <div className="text-sm p-1 min-w-32">
   742→                <strong className="text-[#E31837]">🏁 Destino</strong>
   743→                <br />
   744→                {selectedRoute.destinationStop.name}
   745→                {selectedRoute.destinationStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.destinationStop.city}</span></>}
   746→              </div>
   747→            </Popup>
   748→          </Marker>
   749→        )}
   750→
   751→        {/* Marcador de destino seleccionado */}
   752→        {destinationCoordinates && (
   753→          <Marker
   754→            position={[destinationCoordinates.latitude, destinationCoordinates.longitude]}
   755→            icon={destinationIcon}
   756→          >
   757→            <Popup>
   758→              <div className="text-sm p-1 min-w-32">
   759→                <strong className="text-[#E31837]">🏁 Destino</strong>
   760→                <br />
   761→                {destinationCoordinates.displayName || destinationCoordinates.name}
   762→              </div>
   763→            </Popup>
   764→          </Marker>
   765→        )}
   766→
   767→        {/* Marcador de parada más cercana (solo cuando no hay ruta ni destino seleccionado) */}
   768→        {!selectedRoute && !destinationCoordinates && nearestStop && (
   769→          <Marker
   770→            position={[nearestStop.coordinates.latitude, nearestStop.coordinates.longitude]}
   771→            icon={stopIcon}
   772→          >
   773→            <Popup>
   774→              <div className="text-sm p-1 min-w-32">
   775→                <strong className="text-[#10B981]">🚌 Parada más cercana</strong>
   776→                <br />
   777→                {nearestStop.name}
   778→              </div>
   779→            </Popup>
   780→          </Marker>
   781→        )}
   782→
   783→        {/* Marcadores de paradas de autobús de OpenStreetMap */}
   784→        {busStops && busStops.length > 0 && (
   785→          <>
   786→            {busStops.map((stop) => (
   787→              <Marker
   788→                key={stop.id}
   789→                position={[stop.lat, stop.lon]}
   790→                icon={stopIcon}
   791→              >
   792→                <Popup>
   793→                  <div className="text-sm p-1 min-w-32">
   794→                    <strong className="text-[#10B981]">🚌 Parada</strong>
   795→                    <br />
   796→                    {stop.name || 'Sin nombre'}
   797→                  </div>
   798→                </Popup>
   799→              </Marker>
   800→            ))}
   801→          </>
   802→        )}
   803→      </MapContainer>
   804→    </div>
   805→  )
   806→}
   807→
   808→export default BusMap
