    11→// Función para obtener Leaflet de forma segura (optimizada con cache de promesa)
    12→async function getLeaflet() {
    13→  if (typeof window === 'undefined') return null
    14→
    15→  // Usar cache de promesa para evitar múltiples llamadas simultáneas
    16→  if (!leafletPromise) {
    17→    leafletPromise = import('leaflet').then((L) => {
    18→      // Fix para iconos por defecto de Leaflet en Next.js
    19→      delete (L.Icon.Default.prototype as any)._getIconUrl
    20→      L.Icon.Default.mergeOptions({
    21→        iconRetinaUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png',
    22→        iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png',
    23→        shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png',
    24→      })
    25→      leafletInstance = L
    26→      return L
    27→    })
    28→  }
    29→
    30→  return leafletPromise
    31→}

   548→        })
   549→      }
   550→
   551→      // Segmento 2: Ruta de autobús desde parada de embarque hasta destino
   552→      if (routePath.bus && routePath.bus.length > 0) {
   553→        polylines.push({
   554→          positions: routePath.bus,
   555→          color: '#16a34a', // Verde para autobús
   556→          weight: 6,
   557→        })
   558→      }
   559→    }
   560→    // Fallback: Si no hay ruta con caminos reales pero hay ruta seleccionada, usar líneas rectas
   561→    else if (selectedRoute && userLocation) {
   562→      if (selectedRoute.boardingStop?.coordinates) {
   563→        polylines.push({
   564→          positions: [
   565→            userLocation,
   566→            [selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude]
   567→          ],
   568→          color: '#0052B4',
   569→          weight: 5,
   570→          dashArray: '10, 10',
   571→        })
   572→      }
   573→
   574→      if (selectedRoute.boardingStop?.coordinates && selectedRoute.destinationStop?.coordinates) {
   575→        polylines.push({
   576→          positions: [
   577→            [selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude],
   578→            [selectedRoute.destinationStop.coordinates.latitude, selectedRoute.destinationStop.coordinates.longitude]
   579→          ],
   580→          color: '#16a34a',
   581→          weight: 6,
   582→        })
   583→      }
   584→    }
   585→    // Si no hay ruta seleccionada pero hay destino seleccionado con camino directo
   586→    else if (routePath?.direct && routePath.direct.length > 0) {
   587→      polylines.push({
   588→        positions: routePath.direct,
   589→        color: '#E31837', // Rojo para conexión directa
   590→        weight: 4,
   591→        dashArray: '5, 5',
   592→      })
   593→    }
   594→    // Fallback: Línea directa simple
   595→    else if (userLocation && destinationCoordinates) {
   596→      polylines.push({
   597→        positions: [
   598→          userLocation,
   599→          [destinationCoordinates.latitude, destinationCoordinates.longitude]
   600→        ],
   601→        color: '#E31837',
   602→        weight: 4,
   603→        dashArray: '5, 5',
   604→      })
   605→    }
   606→
   607→    return polylines
   608→  }
   609→
   610→  // No renderizar nada hasta que el componente esté montado en el cliente
   611→  if (!isMounted || !L || !userIcon || !originIcon || !stopIcon || !boardingIcon || !destinationIcon || !busStationIcon) {
   612→    return (
   613→      <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden border border-[#E5E7EB] flex items-center justify-center bg-gray-50">
   614→        <div className="text-center">
   615→          <div className="w-12 h-12 border-4 border-[#E31837] border-t-transparent rounded-full animate-spin mx-auto mb-3"></div>
   616→          <p className="text-sm text-[#6B7280]">Cargando mapa...</p>
   617→        </div>
   618→      </div>
   619→    )
   620→  }
   621→
   622→  return (
   623→    <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden">
   624→      <MapContainer
   625→        center={center}
   626→        zoom={zoom}
   627→        className="w-full h-full"
   628→        scrollWheelZoom={false}
   629→        zoomControl={false}
   630→        ref={mapRef}
   631→      >
   632→        {/* Capa de OpenStreetMap */}
   633→        <TileLayer
   634→          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
   635→          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
   636→        />
   637→
   638→        {/* Controles de zoom personalizados */}
   639→        <ZoomControls />
   640→
   641→        {/* Polilíneas de la ruta seleccionada o conexión directa */}
   642→        {getRoutePolylines().map((polyline, index) => (
   643→          <Polyline
   644→            key={index}
   645→            positions={polyline.positions}
   646→            pathOptions={{
   647→              color: polyline.color,
   648→              weight: polyline.weight,
   649→              dashArray: polyline.dashArray,
   650→              opacity: 0.8,
   651→            }}
   652→          />
   653→        ))}
   654→
   655→        {/* Marcador de origen (ubicación del usuario) */}
   656→        {userLocation && isFinite(userLocation[0]) && isFinite(userLocation[1]) && (
   657→          <Marker
   658→            position={userLocation}
   659→            icon={isTracking ? trackingIcon : (selectedRoute ? userIcon : originIcon)}
   660→          >
   661→            <Popup>
   662→              <div className="text-sm p-1 min-w-32">
   663→                <strong className={isTracking ? "text-[#E31837]" : "text-[#0052B4]"}>
   664→                  {isTracking ? "🚗 Tu posición" : "📍 Origen"}
   665→                </strong>
   666→                <br />
   667→                <span className="text-xs text-[#6B7280]">
   668→                  {isTracking ? "Rastreando en tiempo real" : "Tu ubicación actual"}
   669→                </span>
   670→              </div>
   671→            </Popup>
   672→          </Marker>
   673→        )}
   674→
   675→        {/* Marcadores de paradas de buses de la base de datos (resaltados) */}
   676→        {dbStops && dbStops.length > 0 && (
   677→          <>
   678→            {dbStops.map((stop) => (
   679→              isFinite(stop.latitude) && isFinite(stop.longitude) && (
   680→                <Marker
   681→                  key={`db-${stop.id}`}
   682→                  position={[stop.latitude, stop.longitude]}
   683→                  icon={busStationIcon}
   684→                >
   685→                  <Popup>
   686→                    <div className="text-sm p-1 min-w-40">
   687→                      <div className="flex items-center gap-2 mb-2">
   688→                        <div className="w-6 h-6 bg-[#E31837] rounded-full flex items-center justify-center">
   689→                          <svg viewBox="0 0 24 24" className="w-4 h-4 text-white" fill="currentColor">
   690→                            <path d="M4 16c0 .88.39 1.67 1 2.22V20c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1h8v1c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1.78c.61-.55 1-1.34 1-2.22V6c0-3.5-3.58-4-8-4s-8 .5-8 4v10zm3.5 1c-.83 0-1.5-.67-1.5-1.5S6.67 14 7.5 14s1.5.67 1.5 1.5S8.33 17 7.5 17zm9 0c-.83 0-1.5-.67-1.5-1.5s.67-1.5 1.5-1.5 1.5.67 1.5 1.5-.67 1.5-1.5 1.5zm1.5-6H6V6h12v5z"/>
   691→                          </svg>
   692→                        </div>
   693→                        <strong className="text-[#E31837]">Parada de Bus</strong>
   694→                      </div>
   695→                      <p className="font-semibold text-[#374151]">{stop.name}</p>
   696→                      {stop.city && <p className="text-xs text-[#6B7280] mt-1">{stop.city}</p>}
   697→                      {stop.distance !== undefined && (
   698→                        <p className="text-xs text-[#0052B4] mt-1 font-medium">
   699→                          {stop.distance.toFixed(1)} km de tu ubicación
   700→                        </p>
   701→                      )}
   702→                    </div>
   703→                  </Popup>
   704→                </Marker>
   705→              )
   706→            ))}
   707→          </>
   708→        )}
   709→
   710→        {/* Marcador de parada de embarque */}
   711→        {selectedRoute?.boardingStop?.coordinates &&
   712→         isFinite(selectedRoute.boardingStop.coordinates.latitude) &&
   713→         isFinite(selectedRoute.boardingStop.coordinates.longitude) && (
   714→          <Marker
   715→            position={[selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude]}
   716→            icon={boardingIcon}
   717→          >
   718→            <Popup>
   719→              <div className="text-sm p-1 min-w-32">
   720→                <strong className="text-orange-600">⬆️ Sube aquí</strong>
   721→                <br />
   722→                {selectedRoute.boardingStop.name}
   723→                {selectedRoute.boardingStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.boardingStop.city}</span></>}
   724→              </div>
   725→            </Popup>
   726→          </Marker>
   727→        )}
   728→
   729→        {/* Marcador de destino */}
   730→        {selectedRoute?.destinationStop?.coordinates &&
   731→         isFinite(selectedRoute.destinationStop.coordinates.latitude) &&
   732→         isFinite(selectedRoute.destinationStop.coordinates.longitude) && (
   733→          <Marker
   734→            position={[selectedRoute.destinationStop.coordinates.latitude, selectedRoute.destinationStop.coordinates.longitude]}
   735→            icon={destinationIcon}
   736→          >
   737→            <Popup>
   738→              <div className="text-sm p-1 min-w-32">
   739→                <strong className="text-[#E31837]">🏁 Destino</strong>
   740→                <br />
   741→                {selectedRoute.destinationStop.name}
   742→                {selectedRoute.destinationStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.destinationStop.city}</span></>}
   743→              </div>
   744→            </Popup>
   745→          </Marker>
   746→        )}
   747→
   748→        {/* Marcador de destino seleccionado */}
   749→        {destinationCoordinates &&
   750→         isFinite(destinationCoordinates.latitude) &&
   751→         isFinite(destinationCoordinates.longitude) && (
   752→          <Marker
   753→            position={[destinationCoordinates.latitude, destinationCoordinates.longitude]}
   754→            icon={destinationIcon}
   755→          >
   756→            <Popup>
   757→              <div className="text-sm p-1 min-w-32">
   758→                <strong className="text-[#E31837]">🏁 Destino</strong>
   759→                <br />
   760→                {destinationCoordinates.displayName || destinationCoordinates.name}
   761→              </div>
   762→            </Popup>
   763→          </Marker>
   764→        )}
   765→
   766→        {/* Marcador de parada más cercana (solo cuando no hay ruta ni destino seleccionado) */}
   767→        {!selectedRoute && !destinationCoordinates && nearestStop &&
   768→         isFinite(nearestStop.coordinates.latitude) &&
   769→         isFinite(nearestStop.coordinates.longitude) && (
   770→          <Marker
   771→            position={[nearestStop.coordinates.latitude, nearestStop.coordinates.longitude]}
   772→            icon={stopIcon}
   773→          >
   774→            <Popup>
   775→              <div className="text-sm p-1 min-w-32">
   776→                <strong className="text-[#10B981]">🚌 Parada más cercana</strong>
   777→                <br />
   778→                {nearestStop.name}
   779→              </div>
   780→            </Popup>
   781→          </Marker>
   782→        )}
   783→
   784→        {/* Marcadores de paradas de autobús de OpenStreetMap */}
   785→        {busStops && busStops.length > 0 && (
   786→          <>
   787→            {busStops.map((stop) => (
   788→              isFinite(stop.lat) && isFinite(stop.lon) && (
   789→                <Marker
   790→                  key={stop.id}
   791→                  position={[stop.lat, stop.lon]}
   792→                  icon={stopIcon}
   793→                >
   794→                  <Popup>
   795→                    <div className="text-sm p-1 min-w-32">
   796→                      <strong className="text-[#10B981]">🚌 Parada</strong>
   797→                      <br />
   798→                      {stop.name || 'Sin nombre'}
   799→                    </div>
   800→                  </Popup>
   801→                </Marker>
   802→              )
   803→            ))}
   804→          </>
   805→        )}
   806→      </MapContainer>
   807→    </div>
   808→  )
   809→}
   810→
   811→export default BusMap
