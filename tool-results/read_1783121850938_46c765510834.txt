   415→   415→

   421→   421→    }

   431→   431→    const margin = 0.01 // Aproximadamente 1 km de margen
   432→   432→
   433→   433→    return {
   434→   434→      south: Math.min(...lats) - margin,
   435→   435→      west: Math.min(...lons) - margin,
   436→   436→      north: Math.max(...lats) + margin,
   437→   437→      east: Math.max(...lons) + margin,

   719→   719→  // Calcular distancia entre dos coordenadas usando Haversine
   720→   720→  const calculateDistance = (lat1: number, lon1: number, lat2: number, lon2: number): number => {
   721→   721→    const R = 6371 // Radio de la Tierra en km
   722→   722→    const dLat = (lat2 - lat1) * Math.PI / 180
   723→   723→    const dLon = (lon2 - lon1) * Math.PI / 180
   724→   724→    const a =
   725→   725→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   726→   726→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   727→   727→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   728→   728→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   729→   729→    return R * c
   730→   730→  }
