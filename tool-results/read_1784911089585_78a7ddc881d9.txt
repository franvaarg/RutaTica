   465→    const lons = allPoints.map(p => p[1])
   466→
   467→    const margin = 0.01 // Aproximadamente 1 km de margen
   468→
   469→    return {
   470→      south: Math.min(...lats) - margin,
   471→      west: Math.min(...lons) - margin,
   472→      north: Math.max(...lats) + margin,
   473→      east: Math.max(...lons) + margin,
   474→    }

  1051→    const a =
  1052→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
  1053→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
  1054→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
  1055→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  1056→    return R * c
