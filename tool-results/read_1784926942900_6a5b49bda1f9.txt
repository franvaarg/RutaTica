  1040→  1040→  const calculateDistance = (lat1: number, lon1: number, lat2: number, lon2: number): number => {
  1041→  1041→    const R = 6371 // Radio de la Tierra en km
  1042→  1042→    const dLat = (lat2 - lat1) * Math.PI / 180
  1043→  1043→    const dLon = (lon2 - lon1) * Math.PI / 180
  1044→  1044→    const a =
  1045→  1045→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
  1046→  1046→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
  1047→  1047→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
  1048→  1048→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  1049→  1049→    return R * c
  1050→  1050→  }
