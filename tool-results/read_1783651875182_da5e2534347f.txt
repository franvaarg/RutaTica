    11→// Función para obtener Leaflet de forma segura (optimizada con cache de promesa)
    12→async function getLeaflet() {
    13→  if (typeof window === 'undefined') return null
    14→
    15→  // Usar cache de promesa para evitar múltiples llamadas simultáneas
    16→  if (!leafletPromise) {
    17→    leafletPromise = import('leaflet').then((L) => {
    18→      // Fix para iconos por defecto de Leaflet en Next.js
    19→      delete (L.Icon.Default.prototype as any)._getIconUrl
    20→      L.Icon.Default.mergeOptions({
    21→        iconRetinaUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png',
    22→        iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png',
    23→        shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png',
    24→      })
    25→      leafletInstance = L
    26→      return L
    27→    })
    28→  }
    29→
    30→  return leafletPromise
    31→}

   542→          positions: routePath.bus,
   543→          color: '#16a34a', // Verde para autobús
   544→          weight: 6,
   545→        })
   546→      }
   547→    }
   548→    // Fallback: Si no hay ruta con caminos reales pero hay ruta seleccionada, usar líneas rectas
   549→    else if (selectedRoute && userLocation) {
   550→      if (selectedRoute.boardingStop?.coordinates) {
   551→        polylines.push({
   552→          positions: [
   553→            userLocation,
   554→            [selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude]
   555→          ],
   556→          color: '#0052B4',
   557→          weight: 5,
   558→          dashArray: '10, 10',
   559→        })
   560→      }
   561→
   562→      if (selectedRoute.boardingStop?.coordinates && selectedRoute.destinationStop?.coordinates) {
   563→        polylines.push({
   564→          positions: [
   565→            [selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude],
   566→            [selectedRoute.destinationStop.coordinates.latitude, selectedRoute.destinationStop.coordinates.longitude]
   567→          ],
   568→          color: '#16a34a',
   569→          weight: 6,
   570→        })
   571→      }
   572→    }
   573→    // Si no hay ruta seleccionada pero hay destino seleccionado con camino directo
   574→    else if (routePath?.direct && routePath.direct.length > 0) {
   575→      polylines.push({
   576→        positions: routePath.direct,
   577→        color: '#E31837', // Rojo para conexión directa
   578→        weight: 4,
   579→        dashArray: '5, 5',
   580→      })
   581→    }
   582→    // Fallback: Línea directa simple
   583→    else if (userLocation && destinationCoordinates) {
   584→      polylines.push({
   585→        positions: [
   586→          userLocation,
   587→          [destinationCoordinates.latitude, destinationCoordinates.longitude]
   588→        ],
   589→        color: '#E31837',
   590→        weight: 4,
   591→        dashArray: '5, 5',
   592→      })
   593→    }
   594→
   595→    return polylines
   596→  }
   597→
   598→  // No renderizar nada hasta que el componente esté montado en el cliente
   599→  if (!isMounted || !L || !userIcon || !originIcon || !stopIcon || !boardingIcon || !destinationIcon || !busStationIcon) {
   600→    return (
   601→      <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden border border-[#E5E7EB] flex items-center justify-center bg-gray-50">
   602→        <div className="text-center">
   603→          <div className="w-12 h-12 border-4 border-[#E31837] border-t-transparent rounded-full animate-spin mx-auto mb-3"></div>
   604→          <p className="text-sm text-[#6B7280]">Cargando mapa...</p>
   605→        </div>
   606→      </div>
   607→    )
   608→  }
   609→
   610→  return (
   611→    <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden">
   612→      <MapContainer
   613→        center={center}
   614→        zoom={zoom}
   615→        className="w-full h-full"
   616→        scrollWheelZoom={false}
   617→        zoomControl={false}
   618→        ref={mapRef}
   619→        whenReady={() => {
   620→          // Se dispara una vez que Leaflet terminó su inicialización interna
   621→          // (incluyendo el cálculo del tamaño del contenedor). A partir de aquí
   622→          // es seguro llamar a fitBounds()/setView() sin riesgo de NaN.
   623→          setMapReady(true)
   624→          // Por si el contenedor cambia de tamaño justo después del montaje inicial
   625→          requestAnimationFrame(() => {
   626→            mapRef.current?.invalidateSize()
   627→          })
   628→        }}
   629→      >
   630→        {/* Capa de OpenStreetMap */}
   631→        <TileLayer
   632→          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
   633→          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
   634→        />
   635→
   636→        {/* Controles de zoom personalizados */}
   637→        <ZoomControls />
   638→
   639→        {/* Polilíneas de la ruta seleccionada o conexión directa */}
   640→        {getRoutePolylines().map((polyline, index) => (
   641→          <Polyline
   642→            key={index}
   643→            positions={polyline.positions}
   644→            pathOptions={{
   645→              color: polyline.color,
   646→              weight: polyline.weight,
   647→              dashArray: polyline.dashArray,
   648→              opacity: 0.8,
   649→            }}
   650→          />
   651→        ))}
   652→
   653→        {/* Marcador de origen (ubicación del usuario) */}
   654→        {userLocation && (
   655→          <Marker
   656→            position={userLocation}
   657→            icon={isTracking ? trackingIcon : (selectedRoute ? userIcon : originIcon)}
   658→          >
   659→            <Popup>
   660→              <div className="text-sm p-1 min-w-32">
   661→                <strong className={isTracking ? "text-[#E31837]" : "text-[#0052B4]"}>
   662→                  {isTracking ? "🚗 Tu posición" : "📍 Origen"}
   663→                </strong>
   664→                <br />
   665→                <span className="text-xs text-[#6B7280]">
   666→                  {isTracking ? "Rastreando en tiempo real" : "Tu ubicación actual"}
   667→                </span>
   668→              </div>
   669→            </Popup>
   670→          </Marker>
   671→        )}
   672→
   673→        {/* Marcadores de paradas de buses de la base de datos (resaltados) */}
   674→        {dbStops && dbStops.length > 0 && (
   675→          <>
   676→            {dbStops.map((stop) => (
   677→              <Marker
   678→                key={`db-${stop.id}`}
   679→                position={[stop.latitude, stop.longitude]}
   680→                icon={busStationIcon}
   681→              >
   682→                <Popup>
   683→                  <div className="text-sm p-1 min-w-40">
   684→                    <div className="flex items-center gap-2 mb-2">
   685→                      <div className="w-6 h-6 bg-[#E31837] rounded-full flex items-center justify-center">
   686→                        <svg viewBox="0 0 24 24" className="w-4 h-4 text-white" fill="currentColor">
   687→                          <path d="M4 16c0 .88.39 1.67 1 2.22V20c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1h8v1c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1.78c.61-.55 1-1.34 1-2.22V6c0-3.5-3.58-4-8-4s-8 .5-8 4v10zm3.5 1c-.83 0-1.5-.67-1.5-1.5S6.67 14 7.5 14s1.5.67 1.5 1.5S8.33 17 7.5 17zm9 0c-.83 0-1.5-.67-1.5-1.5s.67-1.5 1.5-1.5 1.5.67 1.5 1.5-.67 1.5-1.5 1.5zm1.5-6H6V6h12v5z"/>
   688→                        </svg>
   689→                      </div>
   690→                      <strong className="text-[#E31837]">Parada de Bus</strong>
   691→                    </div>
   692→                    <p className="font-semibold text-[#374151]">{stop.name}</p>
   693→                    {stop.city && <p className="text-xs text-[#6B7280] mt-1">{stop.city}</p>}
   694→                    {stop.distance !== undefined && (
   695→                      <p className="text-xs text-[#0052B4] mt-1 font-medium">
   696→                        {stop.distance.toFixed(1)} km de tu ubicación
   697→                      </p>
   698→                    )}
   699→                  </div>
   700→                </Popup>
   701→              </Marker>
   702→            ))}
   703→          </>
   704→        )}
   705→
   706→        {/* Marcador de parada de embarque */}
   707→        {selectedRoute?.boardingStop?.coordinates && (
   708→          <Marker
   709→            position={[selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude]}
   710→            icon={boardingIcon}
   711→          >
   712→            <Popup>
   713→              <div className="text-sm p-1 min-w-32">
   714→                <strong className="text-orange-600">⬆️ Sube aquí</strong>
   715→                <br />
   716→                {selectedRoute.boardingStop.name}
   717→                {selectedRoute.boardingStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.boardingStop.city}</span></>}
   718→              </div>
   719→            </Popup>
   720→          </Marker>
   721→        )}
   722→
   723→        {/* Marcador de destino */}
   724→        {selectedRoute?.destinationStop?.coordinates && (
   725→          <Marker
   726→            position={[selectedRoute.destinationStop.coordinates.latitude, selectedRoute.destinationStop.coordinates.longitude]}
   727→            icon={destinationIcon}
   728→          >
   729→            <Popup>
   730→              <div className="text-sm p-1 min-w-32">
   731→                <strong className="text-[#E31837]">🏁 Destino</strong>
   732→                <br />
   733→                {selectedRoute.destinationStop.name}
   734→                {selectedRoute.destinationStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.destinationStop.city}</span></>}
   735→              </div>
   736→            </Popup>
   737→          </Marker>
   738→        )}
   739→
   740→        {/* Marcador de destino seleccionado */}
   741→        {destinationCoordinates && (
   742→          <Marker
   743→            position={[destinationCoordinates.latitude, destinationCoordinates.longitude]}
   744→            icon={destinationIcon}
   745→          >
   746→            <Popup>
   747→              <div className="text-sm p-1 min-w-32">
   748→                <strong className="text-[#E31837]">🏁 Destino</strong>
   749→                <br />
   750→                {destinationCoordinates.displayName || destinationCoordinates.name}
   751→              </div>
   752→            </Popup>
   753→          </Marker>
   754→        )}
   755→
   756→        {/* Marcador de parada más cercana (solo cuando no hay ruta ni destino seleccionado) */}
   757→        {!selectedRoute && !destinationCoordinates && nearestStop && (
   758→          <Marker
   759→            position={[nearestStop.coordinates.latitude, nearestStop.coordinates.longitude]}
   760→            icon={stopIcon}
   761→          >
   762→            <Popup>
   763→              <div className="text-sm p-1 min-w-32">
   764→                <strong className="text-[#10B981]">🚌 Parada más cercana</strong>
   765→                <br />
   766→                {nearestStop.name}
   767→              </div>
   768→            </Popup>
   769→          </Marker>
   770→        )}
   771→
   772→        {/* Marcadores de paradas de autobús de OpenStreetMap */}
   773→        {busStops && busStops.length > 0 && (
   774→          <>
   775→            {busStops.map((stop) => (
   776→              <Marker
   777→                key={stop.id}
   778→                position={[stop.lat, stop.lon]}
   779→                icon={stopIcon}
   780→              >
   781→                <Popup>
   782→                  <div className="text-sm p-1 min-w-32">
   783→                    <strong className="text-[#10B981]">🚌 Parada</strong>
   784→                    <br />
   785→                    {stop.name || 'Sin nombre'}
   786→                  </div>
   787→                </Popup>
   788→              </Marker>
   789→            ))}
   790→          </>
   791→        )}
   792→      </MapContainer>
   793→    </div>
   794→  )
   795→}
   796→
   797→export default BusMap
