   442→  const getRouteBounds = (routePath: RoutePath | null) => {
   443→    if (!routePath) return null
   444→
   445→    const allPoints: [number, number][] = []
   446→
   447→    if (routePath.walking) {
   448→      allPoints.push(...routePath.walking)
   449→    }
   450→    if (routePath.bus) {
   451→      allPoints.push(...routePath.bus)
   452→    }
   453→    if (routePath.direct) {
   454→      allPoints.push(...routePath.direct)
   455→    }
   456→
   457→    if (allPoints.length === 0) return null
   458→
   459→    const lats = allPoints.map(p => p[0])
   460→    const lons = allPoints.map(p => p[1])
   461→
   462→    const margin = 0.01 // Aproximadamente 1 km de margen
   463→
   464→    return {
   465→      south: Math.min(...lats) - margin,
   466→      west: Math.min(...lons) - margin,
   467→      north: Math.max(...lats) + margin,
   468→      east: Math.max(...lons) + margin,
   469→    }
   470→  }

   698→  const calculateDistance = (lat1: number, lon1: number, lat2: number, lon2: number): number => {
   699→    const R = 6371 // Radio de la Tierra en km
   700→    const dLat = (lat2 - lat1) * Math.PI / 180
   701→    const dLon = (lon2 - lon1) * Math.PI / 180
   702→    const a =
   703→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   704→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   705→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   706→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   707→    return R * c
   708→  }

   730→        allPoints[i][0],
   731→        allPoints[i][1],
   732→        allPoints[i + 1][0],
   733→        allPoints[i + 1][1]
   734→      )
   735→    }
