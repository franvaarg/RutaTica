    11→// Función para obtener Leaflet de forma segura (optimizada con cache de promesa)
    12→async function getLeaflet() {
    13→  if (typeof window === 'undefined') return null
    14→
    15→  // Usar cache de promesa para evitar múltiples llamadas simultáneas
    16→  if (!leafletPromise) {
    17→    leafletPromise = import('leaflet').then((L) => {
    18→      // Fix para iconos por defecto de Leaflet en Next.js
    19→      delete (L.Icon.Default.prototype as any)._getIconUrl
    20→      L.Icon.Default.mergeOptions({
    21→        iconRetinaUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png',
    22→        iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png',
    23→        shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png',
    24→      })
    25→      leafletInstance = L
    26→      return L
    27→    })
    28→  }
    29→
    30→  return leafletPromise
    31→}

   537→          <div class="absolute -bottom-6 left-1/2 transform -translate-x-1/2 bg-white px-2 py-1 rounded shadow-md text-xs font-semibold text-[#E31837] whitespace-nowrap">
   538→            Destino
   539→          </div>
   540→        </div>
   541→      `,
   542→      iconSize: [48, 64],
   543→      iconAnchor: [24, 48],
   544→    })
   545→  }, [L])
   546→
   547→  // Crear polilínea para la ruta seleccionada
   548→  const getRoutePolylines = () => {
   549→    const polylines: Array<{
   550→      positions: [number, number][]
   551→      color: string
   552→      weight: number
   553→      dashArray?: string
   554→    }> = []
   555→
   556→    // Si hay ruta seleccionada con caminos reales
   557→    if (selectedRoute && routePath) {
   558→      // Segmento 1: Camino a pie desde ubicación del usuario hasta parada de embarque
   559→      if (routePath.walking && routePath.walking.length > 0) {
   560→        polylines.push({
   561→          positions: routePath.walking,
   562→          color: '#0052B4', // Azul para caminar
   563→          weight: 5,
   564→          dashArray: '10, 10',
   565→        })
   566→      }
   567→
   568→      // Segmento 2: Ruta de autobús usando GTFS shape data o ruta calculada
   569→      if (routePath.bus && routePath.bus.length > 0) {
   570→        polylines.push({
   571→          positions: routePath.bus,
   572→          color: '#16a34a', // Verde para autobús
   573→          weight: 6,
   574→        })
   575→      }
   576→
   577→      // Segmento 3: Camino a pie desde parada de bajada hasta destino final
   578→      if (routePath.walking2 && routePath.walking2.length > 0) {
   579→        polylines.push({
   580→          positions: routePath.walking2,
   581→          color: '#0052B4', // Azul para caminar
   582→          weight: 5,
   583→          dashArray: '10, 10',
   584→        })
   585→      }
   586→    }
   587→    // Ruta directa OSRM (cuando se selecciona un destino antes de planificar)
   588→    else if (routePath?.direct && routePath.direct.length > 2) {
   589→      polylines.push({
   590→        positions: routePath.direct,
   591→        color: '#E31837', // Rojo para conexión directa
   592→        weight: 4,
   593→        dashArray: '5, 5',
   594→      })
   595→    }
   596→    // Sin datos de ruta: NO dibujar líneas rectas — se mostrará "Ruta por definir"
   597→
   598→    return polylines
   599→  }
   600→
   601→  // No renderizar nada hasta que el componente esté montado en el cliente
   602→  if (!isMounted || !L || !userIcon || !originIcon || !customOriginIcon || !stopIcon || !boardingIcon || !destinationIcon || !busStationIcon) {
   603→    return (
   604→      <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden border border-[#E5E7EB] flex items-center justify-center bg-gray-50">
   605→        <div className="text-center">
   606→          <div className="w-12 h-12 border-4 border-[#E31837] border-t-transparent rounded-full animate-spin mx-auto mb-3"></div>
   607→          <p className="text-sm text-[#6B7280]">Cargando mapa...</p>
   608→        </div>
   609→      </div>
   610→    )
   611→  }
   612→
   613→  return (
   614→    <div className="w-full h-full min-h-[300px] rounded-lg overflow-hidden">
   615→      <MapContainer
   616→        center={center}
   617→        zoom={zoom}
   618→        className="w-full h-full"
   619→        scrollWheelZoom={false}
   620→        zoomControl={false}
   621→        ref={mapRef}
   622→        whenReady={() => {
   623→          // Se dispara una vez que Leaflet terminó su inicialización interna
   624→          // (incluyendo el cálculo del tamaño del contenedor). A partir de aquí
   625→          // es seguro llamar a fitBounds()/setView() sin riesgo de NaN.
   626→          setMapReady(true)
   627→          // Por si el contenedor cambia de tamaño justo después del montaje inicial
   628→          requestAnimationFrame(() => {
   629→            mapRef.current?.invalidateSize()
   630→          })
   631→        }}
   632→      >
   633→        {/* Capa de OpenStreetMap */}
   634→        <TileLayer
   635→          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
   636→          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
   637→        />
   638→
   639→        {/* Controles de zoom personalizados */}
   640→        <ZoomControls />
   641→
   642→        {/* Polilíneas de la ruta seleccionada o conexión directa */}
   643→        {getRoutePolylines().map((polyline, index) => (
   644→          <Polyline
   645→            key={index}
   646→            positions={polyline.positions}
   647→            pathOptions={{
   648→              color: polyline.color,
   649→              weight: polyline.weight,
   650→              dashArray: polyline.dashArray,
   651→              opacity: 0.8,
   652→            }}
   653→          />
   654→        ))}
   655→
   656→        {/* Marcador de origen (ubicación del usuario) */}
   657→        {userLocation && (
   658→          <Marker
   659→            position={userLocation}
   660→            icon={isTracking ? trackingIcon : (selectedRoute ? userIcon : originIcon)}
   661→          >
   662→            <Popup>
   663→              <div className="text-sm p-1 min-w-32">
   664→                <strong className={isTracking ? "text-[#E31837]" : "text-[#0052B4]"}>
   665→                  {isTracking ? "🚗 Tu posición" : "📍 Origen"}
   666→                </strong>
   667→                <br />
   668→                <span className="text-xs text-[#6B7280]">
   669→                  {isTracking ? "Rastreando en tiempo real" : "Tu ubicación actual"}
   670→                </span>
   671→              </div>
   672→            </Popup>
   673→          </Marker>
   674→        )}
   675→
   676→        {/* Marcador de origen personalizado */}
   677→        {originCoordinates && (
   678→          <Marker
   679→            position={[originCoordinates.latitude, originCoordinates.longitude]}
   680→            icon={customOriginIcon}
   681→          >
   682→            <Popup>
   683→              <div className="text-sm p-1 min-w-32">
   684→                <strong className="text-[#F59E0B]">🏠 Mi Origen</strong>
   685→                <br />
   686→                <span className="text-xs text-[#6B7280]">{originCoordinates.displayName || originCoordinates.name}</span>
   687→              </div>
   688→            </Popup>
   689→          </Marker>
   690→        )}
   691→
   692→        {/* Marcadores de paradas de buses de la base de datos (resaltados) */}
   693→        {dbStops && dbStops.length > 0 && (
   694→          <>
   695→            {dbStops.map((stop) => (
   696→              <Marker
   697→                key={`db-${stop.id}`}
   698→                position={[stop.latitude, stop.longitude]}
   699→                icon={busStationIcon}
   700→              >
   701→                <Popup>
   702→                  <div className="text-sm p-1 min-w-40">
   703→                    <div className="flex items-center gap-2 mb-2">
   704→                      <div className="w-6 h-6 bg-[#E31837] rounded-full flex items-center justify-center">
   705→                        <svg viewBox="0 0 24 24" className="w-4 h-4 text-white" fill="currentColor">
   706→                          <path d="M4 16c0 .88.39 1.67 1 2.22V20c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1h8v1c0 .55.45 1 1 1h1c.55 0 1-.45 1-1v-1.78c.61-.55 1-1.34 1-2.22V6c0-3.5-3.58-4-8-4s-8 .5-8 4v10zm3.5 1c-.83 0-1.5-.67-1.5-1.5S6.67 14 7.5 14s1.5.67 1.5 1.5S8.33 17 7.5 17zm9 0c-.83 0-1.5-.67-1.5-1.5s.67-1.5 1.5-1.5 1.5.67 1.5 1.5-.67 1.5-1.5 1.5zm1.5-6H6V6h12v5z"/>
   707→                        </svg>
   708→                      </div>
   709→                      <strong className="text-[#E31837]">Parada de Bus</strong>
   710→                    </div>
   711→                    <p className="font-semibold text-[#374151]">{stop.name}</p>
   712→                    {stop.city && <p className="text-xs text-[#6B7280] mt-1">{stop.city}</p>}
   713→                    {stop.distance !== undefined && (
   714→                      <p className="text-xs text-[#0052B4] mt-1 font-medium">
   715→                        {stop.distance.toFixed(1)} km de tu ubicación
   716→                      </p>
   717→                    )}
   718→                  </div>
   719→                </Popup>
   720→              </Marker>
   721→            ))}
   722→          </>
   723→        )}
   724→
   725→        {/* Marcador de parada de embarque */}
   726→        {selectedRoute?.boardingStop?.coordinates && (
   727→          <Marker
   728→            position={[selectedRoute.boardingStop.coordinates.latitude, selectedRoute.boardingStop.coordinates.longitude]}
   729→            icon={boardingIcon}
   730→          >
   731→            <Popup>
   732→              <div className="text-sm p-1 min-w-32">
   733→                <strong className="text-orange-600">⬆️ Sube aquí</strong>
   734→                <br />
   735→                {selectedRoute.boardingStop.name}
   736→                {selectedRoute.boardingStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.boardingStop.city}</span></>}
   737→              </div>
   738→            </Popup>
   739→          </Marker>
   740→        )}
   741→
   742→        {/* Marcador de destino */}
   743→        {selectedRoute?.destinationStop?.coordinates && (
   744→          <Marker
   745→            position={[selectedRoute.destinationStop.coordinates.latitude, selectedRoute.destinationStop.coordinates.longitude]}
   746→            icon={destinationIcon}
   747→          >
   748→            <Popup>
   749→              <div className="text-sm p-1 min-w-32">
   750→                <strong className="text-[#E31837]">🏁 Destino</strong>
   751→                <br />
   752→                {selectedRoute.destinationStop.name}
   753→                {selectedRoute.destinationStop.city && <><br /><span className="text-xs text-[#6B7280]">{selectedRoute.destinationStop.city}</span></>}
   754→              </div>
   755→            </Popup>
   756→          </Marker>
   757→        )}
   758→
   759→        {/* Marcador de destino seleccionado */}
   760→        {destinationCoordinates && (
   761→          <Marker
   762→            position={[destinationCoordinates.latitude, destinationCoordinates.longitude]}
   763→            icon={destinationIcon}
   764→          >
   765→            <Popup>
   766→              <div className="text-sm p-1 min-w-32">
   767→                <strong className="text-[#E31837]">🏁 Destino</strong>
   768→                <br />
   769→                {destinationCoordinates.displayName || destinationCoordinates.name}
   770→              </div>
   771→            </Popup>
   772→          </Marker>
   773→        )}
   774→
   775→        {/* Marcador de parada más cercana (solo cuando no hay ruta ni destino seleccionado) */}
   776→        {!selectedRoute && !destinationCoordinates && nearestStop && (
   777→          <Marker
   778→            position={[nearestStop.coordinates.latitude, nearestStop.coordinates.longitude]}
   779→            icon={stopIcon}
   780→          >
   781→            <Popup>
   782→              <div className="text-sm p-1 min-w-32">
   783→                <strong className="text-[#10B981]">🚌 Parada más cercana</strong>
   784→                <br />
   785→                {nearestStop.name}
   786→              </div>
   787→            </Popup>
   788→          </Marker>
   789→        )}
   790→
   791→        {/* Marcadores de paradas de autobús de OpenStreetMap */}
   792→        {busStops && busStops.length > 0 && (
   793→          <>
   794→            {busStops.map((stop) => (
   795→              <Marker
   796→                key={stop.id}
   797→                position={[stop.lat, stop.lon]}
   798→                icon={stopIcon}
   799→              >
   800→                <Popup>
   801→                  <div className="text-sm p-1 min-w-32">
   802→                    <strong className="text-[#10B981]">🚌 Parada</strong>
   803→                    <br />
   804→                    {stop.name || 'Sin nombre'}
   805→                  </div>
   806→                </Popup>
   807→              </Marker>
   808→            ))}
   809→          </>
   810→        )}
   811→      </MapContainer>
   812→    </div>
   813→  )
   814→}
   815→
   816→export default BusMap
