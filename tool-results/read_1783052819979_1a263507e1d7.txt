   389→   389→

   391→   391→      south: Math.min(...lats) - margin,
   392→   392→      west: Math.min(...lons) - margin,
   393→   393→      north: Math.max(...lats) + margin,
   394→   394→      east: Math.max(...lons) + margin,

   614→   614→  const calculateDistance = (lat1: number, lon1: number, lat2: number, lon2: number): number => {
   615→   615→    const R = 6371 // Radio de la Tierra en km
   616→   616→    const dLat = (lat2 - lat1) * Math.PI / 180
   617→   617→    const dLon = (lon2 - lon1) * Math.PI / 180

   619→   619→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   620→   620→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   621→   621→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   622→   622→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   623→   623→    return R * c
   624→   624→  }
