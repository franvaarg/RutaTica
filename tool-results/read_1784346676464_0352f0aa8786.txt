   378→      if (data.code === 'Ok' && data.routes && data.routes.length > 0) {
   379→        const route = data.routes[0]

   381→        const coordinates = route.geometry.coordinates.map((coord: number[]) => [coord[1], coord[0]]) as [number, number][]
   382→        return coordinates

   384→      return null

   436→  // Calcular los límites del área alrededor de la ruta
   437→  const getRouteBounds = (routePath: RoutePath | null) => {
   438→    if (!routePath) return null
   439→
   440→    const allPoints: [number, number][] = []
   441→
   442→    if (routePath.walking) {
   443→      allPoints.push(...routePath.walking)
   444→    }
   445→    if (routePath.bus) {
   446→      allPoints.push(...routePath.bus)
   447→    }
   448→    if (routePath.walking2) {
   449→      allPoints.push(...routePath.walking2)
   450→    }
   451→    if (routePath.direct) {
   452→      allPoints.push(...routePath.direct)
   453→    }
   454→
   455→    if (allPoints.length === 0) return null
   456→
   457→    const lats = allPoints.map(p => p[0])
   458→    const lons = allPoints.map(p => p[1])
   459→
   460→    const margin = 0.01 // Aproximadamente 1 km de margen
   461→
   462→    return {
   463→      south: Math.min(...lats) - margin,
   464→      west: Math.min(...lons) - margin,
   465→      north: Math.max(...lats) + margin,
   466→      east: Math.max(...lons) + margin,
   467→    }
   468→  }

   852→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   853→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   854→    return R * c
