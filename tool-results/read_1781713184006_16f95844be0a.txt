   158→  const formatAddress = (data: any, useFullAddress: boolean): string => {
   159→    if (!data || !data.address) return ''

   161→    const parts: string[] = []
   162→
   163→    if (useFullAddress) {
   164→      if (addr.road) {
   165→        if (addr.house_number) {
   166→          parts.push(`${addr.road} ${addr.house_number}`)
   167→        } else {
   168→          parts.push(addr.road)
   169→        }
   170→      }
   171→      if (addr.neighbourhood) parts.push(addr.neighbourhood)
   172→      else if (addr.suburb) parts.push(addr.suburb)
   173→      if (addr.village) {
   174→        parts.push(addr.village)
   175→      } else if (addr.town) {
   176→        parts.push(addr.town)
   177→      } else if (addr.city) {
   178→        parts.push(addr.city)
   179→      }
   180→      if (addr.city && !parts.includes(addr.city)) {
   181→        parts.push(addr.city)
   182→      } else if (addr.town && !parts.includes(addr.town)) {
   183→        parts.push(addr.town)
   184→      } else if (addr.city_district) {
   185→        parts.push(addr.city_district)
   186→      }
   187→      if (addr.county) {
   188→        parts.push(addr.county)
   189→      }
   190→      if (addr.state) {
   191→        parts.push(addr.state)
   192→      }
   193→      if (addr.postcode) {
   194→        parts.push(addr.postcode)
   195→      }
   196→    } else {
   197→      if (addr.road) {
   198→        if (addr.house_number) {
   199→          parts.push(`${addr.road} ${addr.house_number}`)
   200→        } else {
   201→          parts.push(addr.road)
   202→        }
   203→      }

   211→        parts.push(addr.city)
   212→      }
   213→    }
   214→
   215→    return parts.join(', ')
   216→  }

   302→  // Función para obtener ruta de OSRM (Open Source Routing Machine)
   303→  const getOSRMRoute = async (start: [number, number], end: [number, number], profile: 'walking' | 'driving' = 'driving') => {
   304→    try {
   305→      const url = `https://router.project-osrm.org/route/v1/${profile}/${start[1]},${start[0]};${end[1]},${end[0]}?overview=full&geometries=geojson`
   306→      const response = await fetch(url)
   307→      const data = await response.json()
   308→
   309→      if (data.code === 'Ok' && data.routes && data.routes.length > 0) {
   310→        const route = data.routes[0]
   311→        // Extraer las coordenadas de la ruta
   312→        const coordinates = route.geometry.coordinates.map((coord: number[]) => [coord[1], coord[0]]) as [number, number][]
   313→        return coordinates
   314→      }
   315→      return null
   316→    } catch (error) {
   317→      console.error('Error al obtener ruta de OSRM:', error)
   318→      return null
   319→    }

   322→  // Función para obtener paradas de autobús de OpenStreetMap usando Overpass API
   323→  const getBusStopsFromOSM = async (bounds: { south: number; west: number; north: number; east: number }) => {
   324→    try {
   325→      setLoadingBusStops(true)
   326→      const query = `
   327→        [out:json][timeout:25];
   328→        (
   329→          node["public_transport"="platform"]["bus"="yes"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});
   330→          node["highway"="bus_stop"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});
   331→        );
   332→        out body;
   333→        >;
   334→        out skel qt;
   335→      `
   336→
   337→      const response = await fetch('https://overpass-api.de/api/interpreter', {
   338→        method: 'POST',
   339→        body: query,
   340→      })
   341→
   342→      const data = await response.json()
   343→
   344→      if (data.elements && data.elements.length > 0) {
   345→        const stops: BusStop[] = data.elements
   346→          .filter((el: any) => el.type === 'node' && el.lat && el.lon)
   347→          .map((el: any) => ({
   348→            id: el.id.toString(),
   349→            name: el.tags?.name || el.tags?.ref || `Parada ${el.id}`,
   350→            lat: el.lat,
   351→            lon: el.lon,
   352→          }))
   353→
   354→        // Limitar a las 50 paradas más cercanas para no saturar el mapa
   355→        return stops.slice(0, 50)
   356→      }
   357→
   358→      return []
   359→    } catch (error) {
   360→      console.error('Error al obtener paradas de autobús:', error)
   361→      return []
   362→    } finally {
   363→      setLoadingBusStops(false)
   364→    }
   365→  }

   367→  // Calcular los límites del área alrededor de la ruta
   368→  const getRouteBounds = (routePath: RoutePath | null) => {
   369→    if (!routePath) return null
   370→
   371→    const allPoints: [number, number][] = []
   372→
   373→    if (routePath.walking) {
   374→      allPoints.push(...routePath.walking)
   375→    }
   376→    if (routePath.bus) {
   377→      allPoints.push(...routePath.bus)
   378→    }
   379→    if (routePath.direct) {
   380→      allPoints.push(...routePath.direct)
   381→    }
   382→
   383→    if (allPoints.length === 0) return null
   384→
   385→    const lats = allPoints.map(p => p[0])
   386→    const lons = allPoints.map(p => p[1])
   387→
   388→    const margin = 0.01 // Aproximadamente 1 km de margen
   389→
   390→    return {
   391→      south: Math.min(...lats) - margin,
   392→      west: Math.min(...lons) - margin,
   393→      north: Math.max(...lats) + margin,
   394→      east: Math.max(...lons) + margin,
   395→    }
   396→  }

   518→          }

   521→          if (route.boardingStop?.coordinates && route.destinationStop?.coordinates) {
   522→            const busRoute = await getOSRMRoute(
   523→              [route.boardingStop.coordinates.latitude, route.boardingStop.coordinates.longitude],
   524→              [route.destinationStop.coordinates.latitude, route.destinationStop.coordinates.longitude],
   525→              'driving'
   526→            )

   528→              newRoutePath.bus = busRoute
   529→            }
   530→          }
   531→
   532→          setRoutePath(newRoutePath)
   533→
   534→          // Ajustar mapa para mostrar toda la ruta
   535→          fitRouteToBounds(newRoutePath)
   536→
   537→          // Obtener paradas de autobús de OpenStreetMap alrededor de la ruta

   539→          if (bounds) {
   540→            const stops = await getBusStopsFromOSM(bounds)
   541→            setBusStops(stops)
   542→          }
   543→        }
   544→
   545→        if (data.routes.length === 0) {
   546→          setError('No se encontraron rutas hacia ese destino. Intenta con otro destino.')
   547→        }
   548→      } else {
   549→        setError(data.error || 'Error al planificar la ruta')
   550→      }
   551→    } catch (error) {
   552→      console.error('Error al planificar ruta:', error)
   553→      setError('Error de conexión. Por favor intenta de nuevo.')
   554→    } finally {
   555→      setPlanning(false)
   556→      setLoadingRoute(false)
   557→    }
   558→  }

   616→    const dLat = (lat2 - lat1) * Math.PI / 180
   617→    const dLon = (lon2 - lon1) * Math.PI / 180
   618→    const a =
   619→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   620→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   621→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   622→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   623→    return R * c
   624→  }
