    32→/**
    33→ * Removes BOM (Byte Order Mark) from the beginning of a string.
    34→ */
    35→function stripBOM(input: string): string {
    36→  if (input.charCodeAt(0) === 0xfeff) {
    37→    return input.slice(1)
    38→  }
    39→  return input
    40→}

    42→/**
    43→ * Parses a single CSV line into an array of fields, handling quoted fields.
    44→ */
    45→function parseCSV(line: string): string[] {
    46→  const fields: string[] = []
    47→  let current = ''
    48→  let inQuotes = false
    49→  let i = 0
    50→
    51→  while (i < line.length) {
    52→    const ch = line[i]
    53→
    54→    if (inQuotes) {
    55→      if (ch === '"') {
    56→        if (i + 1 < line.length && line[i + 1] === '"') {
    57→          current += '"'
    58→          i += 2
    59→          continue
    60→        } else {
    61→          inQuotes = false
    62→          i++
    63→          continue
    64→        }
    65→      } else {
    66→        current += ch
    67→        i++
    68→        continue
    69→      }
    70→    }
    71→
    72→    if (ch === '"') {
    73→      inQuotes = true
    74→      i++
    75→      continue
    76→    }
    77→
    78→    if (ch === ',') {
    79→      fields.push(current.trim())
    80→      current = ''
    81→      i++
    82→      continue
    83→    }
    84→
    85→    current += ch
    86→    i++
    87→  }
    88→
    89→  fields.push(current.trim())
    90→  return fields
    91→}

    93→/**
    94→ * Reads a CSV file and returns an array of rows (each row is an array of string fields).
    95→ * Handles BOM, empty lines, and comment lines starting with #.
    96→ */
    97→function parseCSVFile(filePath: string): string[][] {
    98→  if (!fs.existsSync(filePath)) {
    99→    return []
   100→  }
   101→
   102→  const raw = fs.readFileSync(filePath, 'utf-8')
   103→  const content = stripBOM(raw)
   104→  const lines = content.split(/\r?\n/)
   105→
   106→  const rows: string[][] = []
   107→  let headerFound = false
   108→  let headers: string[] = []
   109→
   110→  for (const line of lines) {
   111→    const trimmed = line.trim()
   112→
   113→    // Skip empty lines
   114→    if (trimmed.length === 0) continue
   115→
   116→    // Skip comment lines starting with #
   117→    if (trimmed.startsWith('#')) continue
   118→
   119→    const fields = parseCSV(trimmed)
   120→
   121→    // First non-empty, non-comment line is the header
   122→    if (!headerFound) {
   123→      headers = fields
   124→      headerFound = true
   125→      continue
   126→    }
   127→
   128→    // Pad or truncate fields to match header length
   129→    while (fields.length < headers.length) {
   130→      fields.push('')
   131→    }
   132→    if (fields.length > headers.length) {
   133→      fields.length = headers.length
   134→    }
   135→
   136→    rows.push(fields)
   137→  }
   138→
   139→  return rows
   140→}

   142→/**
   143→ * Reads a CSV file and returns { headers, rows } where each row is an object
   144→ * mapping header names to field values.
   145→ */
   146→function parseCSVFileAsRecords(filePath: string): { headers: string[]; rows: Record<string, string>[] } {
   147→  if (!fs.existsSync(filePath)) {
   148→    return { headers: [], rows: [] }
   149→  }
   150→
   151→  const raw = fs.readFileSync(filePath, 'utf-8')
   152→  const content = stripBOM(raw)
   153→  const lines = content.split(/\r?\n/)
   154→
   155→  let headers: string[] = []
   156→  const rows: Record<string, string>[] = []
   157→
   158→  for (const line of lines) {
   159→    const trimmed = line.trim()
   160→    if (trimmed.length === 0) continue
   161→    if (trimmed.startsWith('#')) continue
   162→
   163→    const fields = parseCSV(trimmed)
   164→
   165→    if (headers.length === 0) {
   166→      headers = fields
   167→      continue
   168→    }
   169→
   170→    const record: Record<string, string> = {}
   171→    for (let i = 0; i < headers.length; i++) {
   172→      record[headers[i]] = i < fields.length ? fields[i] : ''
   173→    }
   174→
   175→    rows.push(record)
   176→  }
   177→
   178→  return { headers, rows }
   179→}

   185→function parseTime(timeStr: string): string {
   186→  // GTFS times can go past 24:00:00 (e.g., 25:35:00 for trips after midnight).
   187→  // We store them as-is since they're strings in our schema.
   188→  const trimmed = timeStr.trim()
   189→  const match = trimmed.match(/^(\d{1,2}):(\d{2}):(\d{2})$/)
   190→  if (!match) {
   191→    return trimmed
   192→  }
   193→  const h = match[1].padStart(2, '0')
   194→  const m = match[2]
   195→  const s = match[3]
   196→  return `${h}:${m}:${s}`
   197→}

   199→function parseDate(dateStr: string): string {
   200→  // GTFS dates are YYYYMMDD
   201→  const trimmed = dateStr.trim().replace(/-/g, '')
   202→  const match = trimmed.match(/^(\d{4})(\d{2})(\d{2})$/)
   203→  if (!match) {
   204→    return trimmed
   205→  }
   206→  return `${match[1]}${match[2]}${match[3]}`
   207→}

   209→function parseFloat2(str: string): number | null {
   210→  const trimmed = str.trim()
   211→  if (trimmed === '' || trimmed === undefined || trimmed === null) return null
   212→  const num = Number(trimmed)
   213→  if (isNaN(num)) return null
   214→  return num
   215→}
   216→
   217→function parseInt2(str: string): number | null {
   218→  const trimmed = str.trim()
   219→  if (trimmed === '' || trimmed === undefined || trimmed === null) return null
   220→  const num = parseInt(trimmed, 10)
   221→  if (isNaN(num)) return null
   222→  return num
   223→}
   224→
   225→function parseBoolean(str: string): boolean {
   226→  const trimmed = str.trim().toLowerCase()
   227→  return trimmed === '1' || trimmed === 'true' || trimmed === 'yes'
   228→}
   229→
   230→function validateLatitude(lat: number | null): boolean {
   231→  if (lat === null) return false
   232→  return lat >= -90 && lat <= 90
   233→}
   234→
   235→function validateLongitude(lon: number | null): boolean {
   236→  if (lon === null) return false
   237→  return lon >= -180 && lon <= 180
   238→}
   239→
   240→function validateTime(timeStr: string): boolean {
   241→  const trimmed = timeStr.trim()
   242→  // Accept GTFS extended times (can be > 24:00:00)
   243→  const match = trimmed.match(/^(\d{1,3}):(\d{2}):(\d{2})$/)
   244→  if (!match) return false
   245→  const h = parseInt(match[1], 10)
   246→  const m = parseInt(match[2], 10)
   247→  const s = parseInt(match[3], 10)
   248→  if (m < 0 || m > 59) return false
   249→  if (s < 0 || s > 59) return false
   250→  // GTFS allows hours > 24 for overnight trips
   251→  if (h < 0 || h > 99) return false
   252→  return true
   253→}
   254→
   255→function emptyToNull(val: string): string | null {
   256→  const trimmed = val.trim()
   257→  return trimmed === '' ? null : trimmed
   258→}

   267→  recordsCreated: number | null = null,

   276→        recordsCreated,
   277→        recordsUpdated,

   279→        completedAt: new Date(),
   280→      },

   284→  }
   285→}

   287→// ============================================
   288→// Batch Processing Helper
   289→// ============================================
   290→
   291→async function processInBatches<T>(
   292→  items: T[],
   293→  batchSize: number,
   294→  processor: (batch: T[]) => Promise<{ created: number; updated: number }>
   295→): Promise<{ created: number; updated: number }> {
   296→  let totalCreated = 0
   297→  let totalUpdated = 0
   298→
   299→  for (let i = 0; i < items.length; i += batchSize) {
   300→    const batch = items.slice(i, i + batchSize)
   301→    const result = await processor(batch)
   302→    totalCreated += result.created
   303→    totalUpdated += result.updated
   304→  }
   305→
   306→  return { created: totalCreated, updated: totalUpdated }
   307→}

   313→async function importAgencies(): Promise<void> {
   314→  const filename = 'agency.txt'
   315→  const filePath = path.join(GTFS_DIR, filename)
   316→  console.log(`\n📋 Importing ${filename}...`)
   317→
   318→  if (!fs.existsSync(filePath)) {
   319→    console.log(`  ⚠️  File not found: ${filePath}`)
   320→    await createImportLog(filename, 0, 0, 0, 'File not found')
   321→    return
   322→  }
   323→
   324→  try {
   325→    const { rows } = parseCSVFileAsRecords(filePath)
   326→    console.log(`  Found ${rows.length} records`)
   327→
   328→    let created = 0
   329→    let updated = 0
   330→
   331→    const result = await processInBatches(rows, BATCH_SIZE, async (batch) => {
   332→      let bCreated = 0
   333→      let bUpdated = 0
   334→
   335→      for (const row of batch) {
   336→        const agencyId = row['agency_id']
   337→        const name = row['agency_name'] || row['agency_id']
   338→
   339→        if (!agencyId) {
   340→          console.log(`  ⚠️  Skipping agency row without agency_id`)
   341→          continue
   342→        }
   343→
   344→        if (!name) {
   345→          console.log(`  ⚠️  Skipping agency ${agencyId} without name`)
   346→          continue
   347→        }
   348→
   349→        const existing = await prisma.gtfsAgency.findUnique({ where: { agency_id: agencyId } })
   350→
   351→        if (existing) {
   352→          await prisma.gtfsAgency.update({
   353→            where: { agency_id: agencyId },
   354→            data: {
   355→              name: name,
   356→              url: row['agency_url'] || existing.url,
   357→              timezone: row['agency_timezone'] || existing.timezone,
   358→              phone: emptyToNull(row['agency_phone']),
   359→              lang: emptyToNull(row['agency_lang']),
   360→              email: emptyToNull(row['agency_email']),
   361→            },
   362→          })
   363→          bUpdated++
   364→        } else {
   365→          await prisma.gtfsAgency.create({
   366→            data: {
   367→              agency_id: agencyId,
   368→              name: name,
   369→              url: row['agency_url'] || '',
   370→              timezone: row['agency_timezone'] || 'America/Costa_Rica',
   371→              phone: emptyToNull(row['agency_phone']),
   372→              lang: emptyToNull(row['agency_lang']),
   373→              email: emptyToNull(row['agency_email']),
   374→            },
   375→          })
   376→          bCreated++
   377→        }
   378→      }
   379→
   380→      return { created: bCreated, updated: bUpdated }
   381→    })
   382→
   383→    created = result.created
   384→    updated = result.updated
   385→
   386→    console.log(`  ✅ Agencies: ${created} created, ${updated} updated`)
   387→    await createImportLog(filename, rows.length, created, updated)
   388→  } catch (err) {
   389→    const errMsg = err instanceof Error ? err.message : String(err)
   390→    console.error(`  ❌ Error importing ${filename}:`, errMsg)
   391→    await createImportLog(filename, null, null, null, errMsg)
   392→  }
   393→}

   632→    // upsert per row would be very slow with findUnique on all fields)

   634→    await prisma.gtfsStopTime.deleteMany({})

   638→      const batch = rows.slice(i, i + BATCH_SIZE)
   639→      const createData: Prisma.GtfsStopTimeCreateInput[] = []
   640→
   641→      for (const row of batch) {
   642→        const tripId = row['trip_id']
   643→        const stopId = row['stop_id']
   644→
   645→        if (!tripId || !stopId) {
   646→          skipped++
   647→          continue
   648→        }
   649→
   650→        const arrivalTime = parseTime(row['arrival_time'] || '')
   651→        const departureTime = parseTime(row['departure_time'] || '')

   663→        }

   665→        if (!validateTime(departureTime)) {
   666→          console.log(`  ⚠️  Skipping stop_time: invalid departure_time "${row['departure_time']}"`)
   667→          skipped++
   668→          continue
   669→        }

   671→        createData.push({
   672→          trip: { connect: { trip_id: tripId } },
   673→          stop: { connect: { stop_id: stopId } },
   674→          arrival_time: arrivalTime,
   675→          departure_time: departureTime,
   676→          stop_sequence: stopSequence,

  1189→      for (const row of batch) {
  1190→        const routeId = row['route_id']
  1191→        if (!routeId) {
  1192→          bSkipped++
  1193→          continue
  1194→        }
  1195→
  1196→        const color = emptyToNull(row['color'])
  1197→        const textColor = emptyToNull(row['textColor'])
  1198→
  1199→        if (!color && !textColor) {
  1200→          bSkipped++
  1201→          continue
  1202→        }
  1203→
  1204→        const existing = await prisma.gtfsRoute.findUnique({ where: { route_id: routeId } })
  1205→        if (!existing) {
  1206→          console.log(`  ⚠️  Skipping color: route ${routeId} not found`)
  1207→          bSkipped++
  1208→          continue
  1209→        }
  1210→
  1211→        await prisma.gtfsRoute.update({
  1212→          where: { route_id: routeId },
  1213→          data: {
  1214→            ...(color ? { color } : {}),
  1215→            ...(textColor ? { text_color: textColor } : {}),
  1216→          },
  1217→        })
  1218→        bUpdated++
  1219→      }
  1220→
  1221→      return { created: 0, updated: bUpdated }
  1222→    })
  1223→
  1224→    updated = result.updated
  1225→    skipped = rows.length - updated
  1226→
  1227→    console.log(`  ✅ Colores: ${updated} routes updated, ${skipped} skipped`)
  1228→    await createImportLog(filename, rows.length, 0, updated, skipped > 0 ? `${skipped} skipped` : null)
  1229→  } catch (err) {
  1230→    const errMsg = err instanceof Error ? err.message : String(err)
  1231→    console.error(`  ❌ Error importing ${filename}:`, errMsg)
  1232→    await createImportLog(filename, null, null, null, errMsg)
  1233→  }
