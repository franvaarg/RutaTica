   448→    if (!routePath) return null

   451→

   463→    }

   466→
   467→    const lats = allPoints.map(p => p[0])
   468→    const lons = allPoints.map(p => p[1])
   469→
   470→    const margin = 0.01 // Aproximadamente 1 km de margen
   471→
   472→    return {
   473→      south: Math.min(...lats) - margin,
   474→      west: Math.min(...lons) - margin,
   475→      north: Math.max(...lats) + margin,
   476→      east: Math.max(...lons) + margin,
   477→    }
   478→  }

  1041→    const R = 6371 // Radio de la Tierra en km
  1042→    const dLat = (lat2 - lat1) * Math.PI / 180
  1043→    const dLon = (lon2 - lon1) * Math.PI / 180
  1044→    const a =
  1045→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
  1046→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
  1047→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
  1048→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  1049→    return R * c
  1050→  }
