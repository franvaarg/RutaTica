   429→  // Calcular los límites del área alrededor de la ruta
   430→  const getRouteBounds = (routePath: RoutePath | null) => {
   431→    if (!routePath) return null
   432→
   433→    const allPoints: [number, number][] = []
   434→
   435→    if (routePath.walking) {
   436→      allPoints.push(...routePath.walking)
   437→    }

   439→      allPoints.push(...routePath.bus)
   440→    }

   442→      allPoints.push(...routePath.walking2)
   443→    }
   444→    if (routePath.direct) {
   445→      allPoints.push(...routePath.direct)
   446→    }
   447→
   448→    if (allPoints.length === 0) return null
   449→
   450→    const lats = allPoints.map(p => p[0])
   451→    const lons = allPoints.map(p => p[1])
   452→
   453→    const margin = 0.01 // Aproximadamente 1 km de margen
   454→
   455→    return {
   456→      south: Math.min(...lats) - margin,
   457→      west: Math.min(...lons) - margin,
   458→      north: Math.max(...lats) + margin,
   459→      east: Math.max(...lons) + margin,
   460→    }
   461→  }

  1026→  const calculateDistance = (lat1: number, lon1: number, lat2: number, lon2: number): number => {
  1027→    const R = 6371 // Radio de la Tierra en km
  1028→    const dLat = (lat2 - lat1) * Math.PI / 180
  1029→    const dLon = (lon2 - lon1) * Math.PI / 180
  1030→    const a =
  1031→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
  1032→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
  1033→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
  1034→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  1035→    return R * c
  1036→  }
