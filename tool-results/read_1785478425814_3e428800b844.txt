  1027→  const calculateDistance = (lat1: number, lon1: number, lat2: number, lon2: number): number => {
  1028→    const R = 6371 // Radio de la Tierra en km
  1029→    const dLat = (lat2 - lat1) * Math.PI / 180
  1030→    const dLon = (lon2 - lon1) * Math.PI / 180
  1031→    const a =
  1032→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
  1033→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
  1034→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
  1035→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  1036→    return R * c
  1037→  }
