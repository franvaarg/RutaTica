  1026→  const calculateDistance = (lat1: number, lon1: number, lat2: number, lon2: number): number => {
  1027→    const R = 6371 // Radio de la Tierra en km
  1028→    const dLat = (lat2 - lat1) * Math.PI / 180
  1029→    const dLon = (lon2 - lon1) * Math.PI / 180
  1030→    const a =
  1031→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
  1032→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
  1033→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
  1034→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  1035→    return R * c
  1036→  }
