   410→  // Calcular los límites del área alrededor de la ruta
   411→  const getRouteBounds = (routePath: RoutePath | null) => {
   412→    if (!routePath) return null
   413→

   415→
   416→    if (routePath.walking) {
   417→      allPoints.push(...routePath.walking)
   418→    }
   419→    if (routePath.bus) {
   420→      allPoints.push(...routePath.bus)
   421→    }
   422→    if (routePath.direct) {
   423→      allPoints.push(...routePath.direct)
   424→    }
   425→
   426→    if (allPoints.length === 0) return null
   427→
   428→    const lats = allPoints.map(p => p[0])
   429→    const lons = allPoints.map(p => p[1])
   430→
   431→    const margin = 0.01 // Aproximadamente 1 km de margen
   432→
   433→    return {
   434→      south: Math.min(...lats) - margin,
   435→      west: Math.min(...lons) - margin,
   436→      north: Math.max(...lats) + margin,
   437→      east: Math.max(...lons) + margin,
   438→    }
   439→  }

   721→    const R = 6371 // Radio de la Tierra en km
   722→    const dLat = (lat2 - lat1) * Math.PI / 180
   723→    const dLon = (lon2 - lon1) * Math.PI / 180
   724→    const a =
   725→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   726→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   727→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   728→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   729→    return R * c
