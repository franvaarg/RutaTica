   844→   844→  // Calcular distancia entre dos coordenadas usando Haversine
   845→   845→  const calculateDistance = (lat1: number, lon1: number, lat2: number, lon2: number): number => {
   846→   846→    const R = 6371 // Radio de la Tierra en km
   847→   847→    const dLat = (lat2 - lat1) * Math.PI / 180
   848→   848→    const dLon = (lon2 - lon1) * Math.PI / 180
   849→   849→    const a =
   850→   850→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   851→   851→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
   852→   852→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
   853→   853→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   854→   854→    return R * c
   855→   855→  }
