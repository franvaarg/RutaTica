   431→  const getRouteBounds = (routePath: RoutePath | null) => {
   432→    if (!routePath) return null

   435→
   436→    if (routePath.walking) {

   446→      allPoints.push(...routePath.direct)
   447→    }
   448→
   449→    if (allPoints.length === 0) return null
   450→
   451→    const lats = allPoints.map(p => p[0])
   452→    const lons = allPoints.map(p => p[1])
   453→
   454→    const margin = 0.01 // Aproximadamente 1 km de margen
   455→
   456→    return {
   457→      south: Math.min(...lats) - margin,
   458→      west: Math.min(...lons) - margin,
   459→      north: Math.max(...lats) + margin,
   460→      east: Math.max(...lons) + margin,
   461→    }
   462→  }

  1036→    const R = 6371 // Radio de la Tierra en km
  1037→    const dLat = (lat2 - lat1) * Math.PI / 180
  1038→    const dLon = (lon2 - lon1) * Math.PI / 180
  1039→    const a =
  1040→      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
  1041→      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
  1042→      Math.sin(dLon / 2) * Math.sin(dLon / 2)
  1043→    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  1044→    return R * c
  1045→  }
